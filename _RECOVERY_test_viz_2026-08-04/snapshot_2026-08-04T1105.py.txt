def _write_spratt_csv(path: Path, rows: list[tuple[int, float, float]]) -> None:
    """Write a minimal Spratt-format CSV (age_bp, sea_level_m, uncertainty_m)."""
    path.parent.mkdir(parents=True, exist_ok=True)
    # encoding=utf-8 is explicit so the fixture matches the production CSV's
    # encoding on Windows (where open() would otherwise default to cp1252 and
    # write a 0x97 byte for the em-dash that pandas then fails to decode).
    with open(path, "w", encoding="utf-8") as fh:
        fh.write("# Spratt & Lisiecki (2016) — test fixture\n")
        for age_bp, sl, unc in rows:
            fh.write(f"{age_bp},{sl},{unc}\n")


#: What the synthetic fixtures declare in their ``vertical_frame`` GeoTIFF tag.
#: It is deliberately NOT one of the contract's frame names: these rasters are
#: closed-form arithmetic, not a datum, and a fixture that claimed to be "F1" would
#: be the first invented provenance in the file.  What matters for the guard is
#: that the raster says something and the expectation says the same thing.
_SYNTH_FRAME = "SYNTHETIC test fixture datum (closed-form, not a real vertical frame)"


def _write_geotiff(
    path: Path,
    data: np.ndarray,
    bounds: tuple[float, float, float, float] = (6.0, 62.0, 7.0, 63.0),
    crs_epsg: int = 4326,
    nodata: float = -9999.0,
    vertical_frame: str | None = _SYNTH_FRAME,
) -> None:
    """Write a single-band float32 GeoTIFF to *path*.

    *vertical_frame* is written to the ``vertical_frame`` tag, which is what
    ``declared_vertical_frame`` reads and what a ``SillExpectation`` is checked
    against.  Pass ``None`` to write a raster that declares no frame — the
    production GeoTIFFs all declare one, and a frame check that cannot run must not
    look like one that passed.
    """
    path.parent.mkdir(parents=True, exist_ok=True)
    H, W = data.shape
    transform = from_bounds(*bounds, width=W, height=H)
    with rasterio.open(
        path,
        "w",
        driver="GTiff",
        height=H,
        width=W,
        count=1,
        dtype="float32",
        crs=CRS.from_epsg(crs_epsg),
        transform=transform,
        nodata=nodata,
    ) as dst:
        dst.write(data.astype("float32"), 1)
        if vertical_frame is not None:
            dst.update_tags(vertical_frame=vertical_frame)


def _make_coast_gdf(
    sea_levels: list[int] | None = None,
    crs_epsg: int = 4326,
) -> gpd.GeoDataFrame:
    """Build a minimal Paleocoastlines-style GeoDataFrame with one land polygon
    per sea-level step covering roughly half the fixture tile (6–7°E, 62–63°N).
    """
    if sea_levels is None:
        sea_levels = [-120, -80, -40, 0]
    geoms, levels = [], []
    # Land polygon: left half of tile
    land_box = shp_box(6.0, 62.0, 6.5, 63.0)
    for sl in sea_levels:
        geoms.append(land_box)
        levels.append(sl)
    return gpd.GeoDataFrame(
        {"geometry": geoms, "Sea level": levels},
        crs=CRS.from_epsg(crs_epsg),
    )

# ---------------------------------------------------------------------------
# get_sea_level_for_epoch — the half-step tie
#
# The published Spratt & Lisiecki (2016) stack is on a UNIFORM 1 ka grid.  Four
# of the thirteen epochs the Black-Sea bake ships (16.5, 14.5, 9.5, 7.5 ka) land
# exactly halfway between two nodes.  Under the retired nearest-neighbour lookup
# ``(df.age_bp - want).abs().idxmin()`` returned the FIRST row attaining the
# minimum, and the CSV is ascending in age, so every tie broke toward the
# YOUNGER — i.e. the SHALLOWER — node:
#
#     ka    younger node      older node     returned      linear      error
#   16.5    -98.06 @16 ka  -107.30 @17 ka     -98.06    -102.680    +4.620 m
#   14.5    -76.64 @14 ka   -86.57 @15 ka     -76.64     -81.605    +4.965 m
#    9.5    -15.83 @ 9 ka   -24.59 @10 ka     -15.83     -20.210    +4.380 m
#    7.5     -6.11 @ 7 ka    -9.09 @ 8 ka      -6.11      -7.600    +1.490 m
#
# and it did so silently: the miss is exactly 500 yr, which is exactly half a
# step, which is precisely the boundary every guard in the project tests with a
# STRICT inequality.  16.5 and 14.5 ka are the hero frames.
# ---------------------------------------------------------------------------


_PUBLISHED_SPRATT = (
    Path(os.environ.get("PALEO_DATA_ROOT") or _REPO_ROOT.parents[1] / "Paleo-Europe_3D")
    / "data"
    / "raw"
    / "sea_level_curves"
    / "spratt_lisiecki_2016_full.csv"
)

# Published SeaLev_shortPC1 nodes bracketing each shipped half-ka epoch, and the
# value linear interpolation must produce at the midpoint.  Transcribed from
# NOAA/WDS-Paleo study 19982 (Spratt & Lisiecki 2016, Clim. Past 12, 1079-1092);
# ``test_half_ka_expectations_match_the_shipped_published_curve`` proves this
# table against the shipped file rather than trusting the transcription.
_HALF_KA_TRUTH = {
    # epoch_ka: (younger node m, older node m, expected interpolated m)
    16.5: (-98.06, -107.30, -102.680),
    14.5: (-76.64, -86.57, -81.605),
    9.5: (-15.83, -24.59, -20.210),
    7.5: (-6.11, -9.09, -7.600),
}


class TestHalfStepTie:
    @pytest.fixture()
    def uniform_1ka_csv(self, tmp_path: Path) -> Path:
        """A 1-ka grid whose values are strictly decreasing with age."""
        p = tmp_path / "uniform_1ka.csv"
        _write_spratt_csv(p, [(ka * 1000, -10.0 * ka, 1.0) for ka in range(0, 21)])
        return p

    def test_a_half_step_tie_returns_the_midpoint_not_a_neighbour(
        self, uniform_1ka_csv: Path
    ):
        """16.5 ka must not silently become 16 ka.

        Nodes: -160 m at 16 ka, -170 m at 17 ka.  Nearest-neighbour returned
        -160 (the younger, shallower node).  The honest answer is -165.
        """
        sl = get_sea_level_for_epoch(16.5, uniform_1ka_csv)
        assert sl == pytest.approx(-165.0)
        assert sl != pytest.approx(-160.0)  # the younger node it used to pick
        assert sl != pytest.approx(-170.0)  # nor the older one

    def test_the_tie_is_not_merely_broken_the_other_way(
        self, uniform_1ka_csv: Path
    ):
        """A quarter-step must land a quarter of the way, not on a node."""
        assert get_sea_level_for_epoch(16.25, uniform_1ka_csv) == pytest.approx(-162.5)
        assert get_sea_level_for_epoch(16.75, uniform_1ka_csv) == pytest.approx(-167.5)

    def test_every_node_is_returned_bit_exact(self, uniform_1ka_csv: Path):
        """Interpolation must not perturb the published values themselves."""
        for ka in range(0, 21):
            assert get_sea_level_for_epoch(
                float(ka), uniform_1ka_csv
            ) == pytest.approx(-10.0 * ka, abs=0.0, rel=0.0)

    def test_lookup_is_monotone_across_the_tie(self, uniform_1ka_csv: Path):
        """No step discontinuity at the half-step boundary any more."""
        ages = [16.0, 16.1, 16.4, 16.5, 16.6, 16.9, 17.0]
        vals = [get_sea_level_for_epoch(a, uniform_1ka_csv) for a in ages]
        assert all(b < a for a, b in zip(vals, vals[1:])), vals

    @pytest.mark.parametrize("epoch_ka", sorted(_HALF_KA_TRUTH))
    def test_half_ka_expectations_match_the_shipped_published_curve(self, epoch_ka):
        """Prove the transcription in _HALF_KA_TRUTH against the shipped file."""
        if not _PUBLISHED_SPRATT.exists():
            pytest.skip(f"published curve not present: {_PUBLISHED_SPRATT}")
        younger, older, _ = _HALF_KA_TRUTH[epoch_ka]
        lo_ka, hi_ka = int(epoch_ka - 0.5), int(epoch_ka + 0.5)
        assert get_sea_level_for_epoch(
            float(lo_ka), _PUBLISHED_SPRATT
        ) == pytest.approx(younger, abs=1e-9)
        assert get_sea_level_for_epoch(
            float(hi_ka), _PUBLISHED_SPRATT
        ) == pytest.approx(older, abs=1e-9)

    @pytest.mark.parametrize("epoch_ka", sorted(_HALF_KA_TRUTH))
    def test_shipped_half_ka_epochs_no_longer_take_the_shallower_node(self, epoch_ka):
        """The four ties on the real curve, pinned to the interpolated value."""
        if not _PUBLISHED_SPRATT.exists():
            pytest.skip(f"published curve not present: {_PUBLISHED_SPRATT}")
        younger, _older, expected = _HALF_KA_TRUTH[epoch_ka]
        sl = get_sea_level_for_epoch(epoch_ka, _PUBLISHED_SPRATT)
        assert sl == pytest.approx(expected, abs=1e-9)
        assert sl < younger, "the retired lookup returned the shallower node"

    def test_the_thirteen_shipped_epochs_only_move_where_a_tie_existed(self):
        """Regression fence: integer-ka epochs are nodes and must not budge."""
        if not _PUBLISHED_SPRATT.exists():
            pytest.skip(f"published curve not present: {_PUBLISHED_SPRATT}")
        unchanged = {
            22: -125.82, 18: -113.01, 12: -51.05, 11: -35.85, 10: -24.59,
            9: -15.83, 5: 0.00, 2: 4.01, 0: 8.49,
        }
        for ka, want in unchanged.items():
            assert get_sea_level_for_epoch(
                float(ka), _PUBLISHED_SPRATT
            ) == pytest.approx(want, abs=1e-9), ka

class TestSeaLevelRefusals:
    def test_off_the_end_by_exactly_one_step_still_warns(self, tmp_path: Path):
        """The old blind spot: the warning tested ``> 1000`` yr.

        A 0-20 ka curve asked for 21 ka misses by EXACTLY 1000 yr, so the clamp
        to the 20 ka row was silent.  Any extrapolation off the end warns now.
        """
        p = tmp_path / "trunc.csv"
        _write_spratt_csv(p, [(ka * 1000, -10.0 * ka, 1.0) for ka in range(0, 21)])
        with pytest.warns(UserWarning, match="Spratt CSV"):
            sl = get_sea_level_for_epoch(21.0, p)
        assert sl == pytest.approx(-200.0)  # clamped to the 20 ka endpoint

    def test_inside_the_range_never_warns(self, tmp_path: Path):
        """A genuine bracketed lookup is not noisy."""
        p = tmp_path / "ok.csv"
        _write_spratt_csv(p, [(ka * 1000, -10.0 * ka, 1.0) for ka in range(0, 21)])
        with warnings.catch_warnings(record=True) as caught:
            warnings.simplefilter("always")
            get_sea_level_for_epoch(16.5, p)
        assert not [w for w in caught if issubclass(w.category, UserWarning)]

    def test_a_hole_in_the_curve_raises_rather_than_bridging_it(self, tmp_path: Path):
        """Interpolating across 5 ka of missing curve is an invention."""
        p = tmp_path / "coarse.csv"
        _write_spratt_csv(p, [(0, 0.0, 1.0), (5000, -10.0, 1.0), (25000, -120.0, 1.0)])
        with pytest.raises(SeaLevelCurveResolutionError) as exc:
            get_sea_level_for_epoch(16.5, p)
        assert "5000" in str(exc.value) and "25000" in str(exc.value)

    def test_a_node_inside_a_hole_is_still_answerable(self, tmp_path: Path):
        """The refusal is about interpolating, not about coarse curves per se."""
        p = tmp_path / "coarse.csv"
        _write_spratt_csv(p, [(0, 0.0, 1.0), (5000, -10.0, 1.0), (25000, -120.0, 1.0)])
        assert get_sea_level_for_epoch(5.0, p) == pytest.approx(-10.0)
        assert get_sea_level_for_epoch(25.0, p) == pytest.approx(-120.0)

    def test_the_gap_budget_is_caller_controllable(self, tmp_path: Path):
        """A caller who owns a 5-ka curve can say so explicitly, in writing."""
        p = tmp_path / "coarse.csv"
        _write_spratt_csv(p, [(0, 0.0, 1.0), (5000, -10.0, 1.0)])
        assert get_sea_level_for_epoch(
            2.5, p, max_node_gap_yr=5_000.0
        ) == pytest.approx(-5.0)

    def test_unsorted_and_duplicated_rows_do_not_change_the_answer(self, tmp_path: Path):
        """np.interp is undefined on unsorted x; the loader must not rely on order."""
        p = tmp_path / "shuffled.csv"
        _write_spratt_csv(
            p,
            [(17000, -170.0, 1.0), (15000, -150.0, 1.0),
             (16000, -160.0, 1.0), (16000, -160.0, 1.0)],
        )
        assert get_sea_level_for_epoch(16.5, p) == pytest.approx(-165.0)

def test_coastline_step_selection_stays_nearest_and_never_interpolates():
    """``filter_paleocostlines_for_epoch`` keys a CATEGORICAL set of polygons.

    Its "Sea level" values are labels of discrete published coastline layers,
    not samples of a continuous function: there is no polygon halfway between
    the -120 m and the -80 m step, and inventing a level of -100 would select
    nothing at all.  It must keep snapping to an EXISTING step.
    """
    gdf = _make_coast_gdf([-120, -80, -40, 0])
    _, step = filter_paleocostlines_for_epoch(gdf, -100.0)  # exact half-step tie
    assert step in (-120, -80), "must return a real published step, not -100"
    assert isinstance(step, int)

class TestRasterizeLandMask:
    """Tests for rasterize_land_mask using in-memory data only."""

    # Fixture tile: 6–7°E, 62–63°N, 64×64 px
    _bounds = rasterio.coords.BoundingBox(left=6.0, bottom=62.0, right=7.0, top=63.0)
    _crs = CRS.from_epsg(4326)
    _H, _W = 64, 64
    _transform = from_bounds(6.0, 62.0, 7.0, 63.0, width=_W, height=_H)

    def test_half_tile_land_mask(self):
        """Land polygon covering left half → roughly 50 % True pixels."""
        gdf = _make_coast_gdf(sea_levels=[-120])
        land = gdf[gdf["Sea level"] == -120].copy()
        mask = rasterize_land_mask(
            land, self._transform, self._H, self._W, self._crs, self._bounds
        )
        assert mask is not None
        assert mask.dtype == bool
        land_frac = mask.sum() / mask.size
        assert 0.35 < land_frac < 0.65, f"Expected ~50 % land, got {land_frac:.2%}"

    def test_none_gdf_returns_none(self):
        mask = rasterize_land_mask(
            None, self._transform, self._H, self._W, self._crs, self._bounds
        )
        assert mask is None

    def test_nonintersecting_polygon_returns_none(self):
        """Polygon entirely outside the scene extent → None."""
        far = gpd.GeoDataFrame(
            {"geometry": [shp_box(50.0, 0.0, 51.0, 1.0)], "Sea level": [-120]},
            crs=self._crs,
        )
        mask = rasterize_land_mask(
            far, self._transform, self._H, self._W, self._crs, self._bounds
        )
        assert mask is None

class TestBuildEpochMosaic:
    """Tests using synthetic single-tile mosaics (no real data)."""

    @pytest.fixture()
    def single_tile(self, tmp_path: Path) -> Path:
        """One 64×64 tile at 62–63°N, 6–7°E with mixed land/ocean."""

    def two_tiles(self, tmp_path: Path) -> list[Path]:
        """Two adjacent 64×64 tiles."""
        rng = np.random.default_rng(1)
        paths = []
        for col, (lon0, lon1) in enumerate([(6.0, 7.0), (7.0, 8.0)]):
            data = rng.uniform(-50.0, 200.0, (64, 64)).astype("float32")
            p = tmp_path / f"tile{col}.tif"

    def test_ice_array_same_shape_as_dem(self, single_tile: Path):
        cfg = RenderConfig(max_px=64)

class TestSeaLevelOverride:
    """The keyword-only ``sea_level_override_m`` must replace the global Spratt
    value for the coastline-step selection while leaving the default path
    (override=None) byte-for-byte identical to today's behaviour.

    Scientific motivation: inside a sill-dammed basin (the Black Sea was an
    isolated lake until ~9 ka) the global eustatic curve is the WRONG sea level
    — the basin must be driven by its own cited level curve.
    """

    @pytest.fixture(autouse=True)
    def setup(self, tmp_path: Path):
        self.tmp = tmp_path

        # Spratt CSV: 21 ka → -125 m.  We then override with -40 m so the
        # selected coastline step is unambiguously different.
        self.csv = tmp_path / "spratt.csv"
        rows = [(ka * 1000, float(-ka * 6), 2.0) for ka in range(0, 27)]
        _write_spratt_csv(self.csv, rows)  # 21 ka → -126 m

        rng = np.random.default_rng(7)
        data = rng.uniform(-30.0, 300.0, (32, 32)).astype("float32")
        self.tile = tmp_path / "N62E006_fusion.tif"
        _write_geotiff(self.tile, data)

        # Coastline steps far apart so nearest-step picks are unambiguous.
        shp = tmp_path / "coast" / "Paleocoastlines.shp"
        shp.parent.mkdir(parents=True, exist_ok=True)
        _make_coast_gdf([-120, -80, -40, 0]).to_file(shp)
        self.shp = shp

        self.outdir = tmp_path / "out"
        self.cfg = RenderConfig(max_px=32, dpi=30, fig_size_in=3.0)

    def test_override_selects_step_nearest_override_not_spratt(self):
        """Unit-level: the engine must feed the OVERRIDE (not the Spratt value)
        into filter_paleocostlines_for_epoch.

        Spratt(21 ka) = -126 m  → nearest step -120.
        override      = -40 m   → nearest step  -40.
        These differ, proving the override is what drives selection.
        """
        gdf = _make_coast_gdf([-120, -80, -40, 0])

        spratt_sl = get_sea_level_for_epoch(21.0, self.csv)
        _, step_default = filter_paleocostlines_for_epoch(gdf, spratt_sl)
        _, step_override = filter_paleocostlines_for_epoch(gdf, -40.0)

        assert step_default == -120
        assert step_override == -40
        assert step_default != step_override

    def test_render_with_override_differs_from_default(self):
        """End-to-end: a render WITH override=-40 must encode a different
        coastline step than the default (Spratt) render — visible in the PNG
        bytes — while both still produce a valid file.
        """
        png_default = render_paleocostline_epoch(
            epoch_ka=21.0,
            tile_paths={str(self.tile): str(self.tile)},
            sea_level_csv=self.csv,
            paleocoastlines_shp=self.shp,
            output_dir=self.outdir / "default",
            config=self.cfg,
        )
        png_override = render_paleocostline_epoch(
            epoch_ka=21.0,
            tile_paths={str(self.tile): str(self.tile)},
            sea_level_csv=self.csv,
            paleocoastlines_shp=self.shp,
            output_dir=self.outdir / "override",
            config=self.cfg,
            sea_level_override_m=-40.0,
        )
        assert png_default.exists()
        assert png_override.exists()
        # Different coastline step → different rasterized land mask → different
        # rendered pixels.
        assert png_default.read_bytes() != png_override.read_bytes()

    def test_override_none_is_backward_compatible(self):
        """override=None (default) must select the SAME step as a render that
        never passed the kwarg — i.e. the global Spratt path is unchanged.
        """
        spratt_sl = get_sea_level_for_epoch(21.0, self.csv)
        gdf = _make_coast_gdf([-120, -80, -40, 0])
        _, step_spratt = filter_paleocostlines_for_epoch(gdf, spratt_sl)

        png = render_paleocostline_epoch(
            epoch_ka=21.0,
            tile_paths={str(self.tile): str(self.tile)},
            sea_level_csv=self.csv,
            paleocoastlines_shp=self.shp,
            output_dir=self.outdir,
            config=self.cfg,
            sea_level_override_m=None,
        )
        assert png.exists()
        assert step_spratt == -120  # unchanged default selection

# Synthetic two-bowl world -------------------------------------------------
# Two circular cones rising from -100 m at their apices, capped at +300 m:
#
#     dem = min(-100 + 4*dist_to_A, -100 + 4*dist_to_B, 300)
#
# so the saddle between them sits at the midpoint, where both distances are 35
# cells: -100 + 4*35 = +40.00 m.  That number is ANALYTIC, not measured off a
# picture, which is what makes it usable as an expected value for the sill.
_BASIN_H, _BASIN_W = 60, 160
_BASIN_BOUNDS = (0.0, 0.0, 16.0, 6.0)          # west, south, east, north
_BASIN_CELL_DEG = 0.1
_A_RC, _B_RC = (30, 40), (30, 110)             # bowl apices (row, col)
_SADDLE_M = 40.0                               # -100 + 4 * 35


def _basin_dem() -> np.ndarray:
    r, c = np.indices((_BASIN_H, _BASIN_W))
    d_a = np.hypot(r - _A_RC[0], c - _A_RC[1])
    d_b = np.hypot(r - _B_RC[0], c - _B_RC[1])
    dem = np.minimum(-100.0 + 4.0 * d_a, -100.0 + 4.0 * d_b)
    return np.minimum(dem, 300.0)


def _basin_transform():
    return from_bounds(*_BASIN_BOUNDS, width=_BASIN_W, height=_BASIN_H)


def _rc_to_lonlat(rc: tuple[int, int]) -> tuple[float, float]:
    row, col = rc
    west, south, east, north = _BASIN_BOUNDS
    return (
        west + (col + 0.5) * _BASIN_CELL_DEG,
        north - (row + 0.5) * _BASIN_CELL_DEG,
    )


def _seeds(level_a: float, level_b: float) -> list[BasinSeed]:
    lon_a, lat_a = _rc_to_lonlat(_A_RC)
    lon_b, lat_b = _rc_to_lonlat(_B_RC)
    return [
        BasinSeed(name="west", lon=lon_a, lat=lat_a, level_m=level_a),
        BasinSeed(name="east", lon=lon_b, lat=lat_b, level_m=level_b),
    ]


def _one_bowl_dem() -> np.ndarray:
    """The two-bowl world with the SECOND bowl removed.

    Needed because ``UnseededWaterBodyError`` now fires on any painted component
    that holds no seed (see its docstring).  In the two-bowl world a single seed at
    the west apex paints the east bowl's floor as well — it is below the west
    basin's level and the ownership flood labels it "west" — and that is exactly the
    Aegean defect at fixture scale, so the frame is correctly refused.  A test about
    the SILL guard not taxing single-basin frames needs a world with one basin in
    it, not a world where the refusal under test is a different one.
    """
    r, c = np.indices((_BASIN_H, _BASIN_W))
    d_a = np.hypot(r - _A_RC[0], c - _A_RC[1])
    return np.minimum(-100.0 + 4.0 * d_a, 300.0)


# ---------------------------------------------------------------------------
# The sill guard is mandatory, so every multi-basin call must say something
# ---------------------------------------------------------------------------
#
# ``assign_basins`` used to accept ``sill_expectations=None`` and validate nothing,
# so the two-bowl and three-bowl fixtures below said nothing about their sills and
# nothing checked them.  They are ANALYTIC worlds -- the saddle of the two-bowl DEM
# is +40.00 m by construction, not by measurement -- so stating the expectation
# costs nothing and is exactly what a real caller has to do.  Where the fixture
# deliberately makes a saddle unreachable, the pair is declared UnmeasuredSill with
# the reason, which is the only other thing the guard accepts.

_SADDLE_LON, _SADDLE_LAT = _rc_to_lonlat((30, 75))


def _expect(value_m: float, *, lon: float | None = None,
            lat: float | None = None) -> SillExpectation:
    """The two-bowl world's saddle, stated the way a production caller states one."""
    return SillExpectation(
        value_m=value_m,
        frame=_SYNTH_FRAME,
        source="tests/test_viz.py::_basin_dem — closed form, no measurement involved",
        measured_by=(
            "analytic: dem = min(-100 + 4*d_A, -100 + 4*d_B) makes the divide "
            "equidistant at 35 cells, so the crest is -100 + 4*35"
        ),
        lon=_SADDLE_LON if lon is None else lon,
        lat=_SADDLE_LAT if lat is None else lat,
    )


def _two_bowl_checks(value_m: float | None = None, *, reason: str = "") -> dict:
    key = ("east", "west")
    return {key: _expect(value_m) if value_m is not None else UnmeasuredSill(reason)}


def _assign2(
    dem: np.ndarray,
    seeds: list[BasinSeed],
    *,
    expect_m: float | None = _SADDLE_M,
    unmeasured_because: str = "",
    **kwargs,
):
    """``assign_basins`` on the two-bowl world with its analytic saddle stated.

    ``expect_m`` defaults to ``_SADDLE_M`` = +40.00 m, which is what the fixture's
    own arithmetic produces on the UNSMOOTHED array; a caller that smooths first
    passes the smoothed value, and one whose saddle the fixture puts out of reach
    passes ``expect_m=None`` with a reason.  There is no path through this helper
    that checks nothing.
    """
    return assign_basins(
        dem,
        _basin_transform(),
        seeds,
        sill_expectations=_two_bowl_checks(expect_m, reason=unmeasured_because),
        dem_frame=_SYNTH_FRAME,
        **kwargs,
    )


class TestBasinAssignment:
    """``assign_basins`` -- hydrologically honest membership + observed spill."""

    def test_membership_follows_the_drainage_divide_not_a_bounding_box(self):
        """Every cell must be owned by the bowl it drains into.  In this world
        that divide is the column-75 midline, and it is derivable analytically:
        a cell is closer to A iff it is west of the midline.
        """
        dem = _basin_dem()
        field = _assign2(dem, _seeds(-40.0, -60.0))

        west_id = field.names.index("west") + 1
        east_id = field.names.index("east") + 1
        assert field.basin_id[_A_RC] == west_id
        assert field.basin_id[_B_RC] == east_id

        # Interior of each bowl, well clear of the divide, is owned outright.
        assert field.basin_id[30, 45] == west_id
        assert field.basin_id[30, 105] == east_id
        # A cell EAST of the divide but at the SAME longitude band a bounding
        # box would hand to the west basin still drains east.
        assert field.basin_id[30, 80] == east_id
        assert field.basin_id[30, 70] == west_id

    def test_sill_elevation_is_measured_from_the_terrain(self):
        """The saddle between the two bowls is analytically +40.00 m; the
        assignment must recover it from the DEM without being told."""
        dem = _basin_dem()
        # Ceiling must clear the saddle, else the flood cannot see it.
        field = _assign2(dem, _seeds(-40.0, -60.0))
        assert field.sill_m[("east", "west")] == pytest.approx(_SADDLE_M, abs=0.6)
        lon, lat = field.sill_lonlat[("east", "west")]
        assert lon == pytest.approx(7.55, abs=0.25)   # column 75
        assert lat == pytest.approx(2.95, abs=0.25)   # row 30

    def test_sill_above_the_flood_ceiling_is_reported_as_unknown_not_guessed(self):
        """If both levels sit so far below the saddle that the flood never
        reaches it, the sill is NaN -- never a plausible-looking number."""
        dem = _basin_dem()
        field = _assign2(
            dem,
            _seeds(-40.0, -60.0),
            expect_m=None,
            unmeasured_because=(
                "this fixture deliberately caps the flood 30 m below the +40.00 m "
                "saddle, so the pair has no measurable crest to compare against"
            ),
            ceiling_margin_m=10.0,
        )
        assert np.isnan(field.sill_m[("east", "west")])
        assert field.spilling is False

    def test_not_spilling_when_both_levels_are_below_the_sill(self):
        dem = _basin_dem()
        field = _assign2(dem, _seeds(20.0, -20.0))
        assert field.spilling is False
        assert field.connected_groups == (("east",), ("west",))
        assert not field.corridor_mask.any()
        assert field.corridor_mode == "none"

    def test_spilling_is_observed_when_the_donor_level_clears_the_sill(self):
        dem = _basin_dem()
        field = _assign2(dem, _seeds(60.0, -20.0))
        assert field.spilling is True
        assert field.connected_groups == (("east", "west"),)

    def test_each_basin_is_thresholded_against_its_own_level(self):
        dem = _basin_dem()
        field = _assign2(dem, _seeds(20.0, -20.0))
        west_id = field.names.index("west") + 1
        east_id = field.names.index("east") + 1
        assert field.water_level_m[30, 45] == pytest.approx(20.0)
        assert field.water_level_m[30, 105] == pytest.approx(-20.0)
        assert field.water_mask[_A_RC] and field.water_mask[_B_RC]
        # dem == 0 m happens at distance 25 cells from each apex.
        assert field.basin_id[30, 65] == west_id and dem[30, 65] == pytest.approx(0.0)
        assert field.water_mask[30, 65]           # 0 < +20 -> west water
        assert field.basin_id[30, 85] == east_id and dem[30, 85] == pytest.approx(0.0)
        assert not field.water_mask[30, 85]       # 0 > -20 -> east dry

    def test_determinism(self):
        dem = _basin_dem()
        a = _assign2(dem, _seeds(60.0, -20.0))
        b = _assign2(dem, _seeds(60.0, -20.0))
        np.testing.assert_array_equal(a.basin_id, b.basin_id)
        np.testing.assert_allclose(a.water_level_m, b.water_level_m, equal_nan=True)

    def test_seed_off_the_raster_raises(self):
        dem = _basin_dem()
        with pytest.raises(ValueError, match="outside"):
            assign_basins(
                dem,
                _basin_transform(),
                [BasinSeed("nowhere", lon=99.0, lat=0.0, level_m=0.0)],
            )

    def test_seed_that_is_not_submerged_at_its_own_level_raises(self):
        """A basin seed must be under its own water.  A seed on dry land silently
        produces an empty basin, which is exactly the class of quiet wrongness
        this project refuses."""
        dem = _basin_dem()
        with pytest.raises(ValueError, match="above its own level"):
            assign_basins(dem, _basin_transform(), _seeds(-40.0, -200.0))

    def test_duplicate_basin_names_raise(self):
        dem = _basin_dem()
        lon_a, lat_a = _rc_to_lonlat(_A_RC)
        lon_b, lat_b = _rc_to_lonlat(_B_RC)
        with pytest.raises(ValueError, match="unique"):
            assign_basins(
                dem,
                _basin_transform(),
                [
                    BasinSeed("same", lon_a, lat_a, -40.0),
                    BasinSeed("same", lon_b, lat_b, -60.0),
                ],
            )


class TestSpillCorridor:
    """The water surface between two basins at different levels is a GRADIENT."""

    def test_corridor_is_a_ramp_not_a_step(self):
        dem = _basin_dem()
        field = _assign2(dem, _seeds(60.0, -20.0))
        assert field.corridor_mode == "ramp"
        assert field.corridor_mask.any()

        ramp = field.water_level_m[field.corridor_mask]
        # Strictly between the two basin levels, and a genuine spread -- a step
        # would put every corridor cell at one of the two endpoint values.
        assert ramp.min() > -20.0
        assert ramp.max() <= 60.0 + 1e-9
        assert ramp.max() - ramp.min() > 5.0
        # Endpoint-exact at the donor end: the first corridor cells past the
        # sill stand essentially at the donor level.
        assert ramp.max() > 55.0

    def test_the_ramp_does_not_fill_the_receiving_basin_to_the_donor_level(self):
        """The failure mode this exists to prevent: flooding the whole receiving
        basin to the donor's surface because ``dem < donor_level`` is true there.
        """
        dem = _basin_dem()
        field = _assign2(dem, _seeds(60.0, -20.0))
        far = (30, 135)                      # 25 cells east of the east apex
        assert dem[far] == pytest.approx(0.0)
        assert dem[far] < 60.0                # a naive donor-level fill WOULD flood it
        assert not field.water_mask[far]      # the ramp does not
        assert field.water_level_m[far] < 0.0

    def test_donor_basin_keeps_its_own_flat_surface(self):
        dem = _basin_dem()
        field = _assign2(dem, _seeds(60.0, -20.0))
        west_id = field.names.index("west") + 1
        west_water = (
            field.water_mask & (field.basin_id == west_id) & (~field.corridor_mask)
        )
        assert west_water.any()
        assert np.allclose(field.water_level_m[west_water], 60.0)

    def test_step_mode_marks_the_corridor_instead_of_pretending(self):
        dem = _basin_dem()

    def test_more_than_one_spilling_pair_raises_rather_than_guessing(self):

                },

class TestDemThresholdOceanAcceptsAField:
    """``dem_threshold_ocean`` must take a per-cell water surface, and must be
    bit-identical to the scalar call when that surface is constant."""

    def test_constant_field_equals_the_scalar_call(self):
        rng = np.random.default_rng(3)
        dem = rng.uniform(-120.0, 40.0, (40, 40))
        nodata = np.zeros((40, 40), dtype=bool)
        scalar = dem_threshold_ocean(dem, nodata, -30.0)
        field = dem_threshold_ocean(dem, nodata, np.full((40, 40), -30.0))
        np.testing.assert_array_equal(scalar, field)

    def test_a_varying_field_thresholds_per_cell(self):
        dem = np.full((40, 40), -40.0)
        nodata = np.zeros((40, 40), dtype=bool)
        level = np.full((40, 40), -80.0)
        level[:, 20:] = 0.0
        out = dem_threshold_ocean(dem, nodata, level)
        assert not out[4:-4, :16].any()      # -40 is above the -80 surface
        assert out[4:-4, 24:-4].all()        # -40 is below the 0 m surface

# ---------------------------------------------------------------------------
# The sill the FRAME can resolve vs the sill the TERRAIN has
# ---------------------------------------------------------------------------
#
# Measured on the real Khvalynian raster, Manych point sill 44.119E / 45.773N,
# through build_epoch_mosaic itself:
#
#                                             max_px=2000   max_px=4000
#   native 15 arcsec, untouched ............. +23.00 m       +23.00 m  <- the terrain
#   + bilinear downsample to the mosaic ..... +24.00 m       +25.00 m
#   + the seam blend as it stood 2026-08-01 . +40.76 m       +30.57 m  <- the frame
#
# The last row is gone: that blend ran along every whole degree line whether or
# not a tile boundary was there, and the single-file inputs both bakes pass have
# no seams at all (see TestSeamBlendDoesNotMoveTheManychCrest).  What is left is
# the honest cost of downsampling a spillway narrower than the working pixel,
# +1.00 m and +2.00 m, and the renderer's own dem_sm gaussian on top of it.
#
# The 16.76 m that used to ride along was the dangerous part: at a +35 m
# Khvalynian highstand it flips the frame's spill verdict from SPILLING to
# endorheic -- silently, and in the direction that deletes the subject of the
# animation.  `spilling` is only as honest as the DEM it is observed on, so a
# caller who has measured the sill at native resolution can state it and have the
# frame refuse to disagree -- which is the guard that stays put whether or not any
# one smoothing step is currently behaving.

from paleoeurope.viz.paleocostline_render import SillResolutionError  # noqa: E402

        assert field.sill_expectation_status[("east", "west")].startswith("CHECKED:")

    def test_an_expectation_for_a_sill_above_the_ceiling_raises(self):

# ---------------------------------------------------------------------------
# Three basins in a line: connectivity must be a DONOR-level flood
# ---------------------------------------------------------------------------
#
# The hero frame is a THREE-basin configuration -- Caspian, Pontic, Mediterranean
# -- and the pairwise test ``dem < max(level_i, level_j)`` manufactures a
# connection across it.  Flooding to the pairwise max floods every basin in
# BETWEEN to the donor's level as well, so a Caspian at +35 m "reaches" the
# Mediterranean straight through a Pontic basin standing 65 m lower, whatever the
# Bosphorus sill does.  The frame then reports two spilling pairs and
# ``corridor="ramp"`` refuses to draw it at all.
#
# The world below is that configuration reduced to its analytic bones: a channel
# with three bowls at -100 m separated by two crests, +20 m between caspian and
# pontic and +5 m between pontic and mediterranean.  Every number in it is a
# control point of the profile, not something read off a picture.
#
#   caspian  +35 m  clears the +20 m crest  -> spills into pontic
#   pontic   -30 m  is BELOW the +5 m crest -> does not spill into mediterranean
#   caspian  +35 m  is above BOTH crests    -> the old pairwise test connected it
#                                              to the mediterranean as well

import itertools  # noqa: E402

_CHAIN_H, _CHAIN_W = 40, 180
_CHAIN_BOUNDS = (0.0, 0.0, 18.0, 4.0)          # 0.1 deg cells, both axes
_CHAIN_CELL_DEG = 0.1
#                     col    elevation
_CHAIN_PROFILE = ((0, 300.0), (30, -100.0), (60, 20.0), (90, -100.0),
                  (120, 5.0), (150, -100.0), (179, 300.0))
_CHAIN_SEED_COL = {"caspian": 30, "pontic": 90, "mediterranean": 150}
_CHAIN_SEED_ROW = 20
_CHAIN_CREST_CASPIAN_PONTIC_M = 20.0
_CHAIN_CREST_PONTIC_MED_M = 5.0
_CHAIN_LEVELS = {"caspian": 35.0, "pontic": -30.0, "mediterranean": -45.0}


def _chain_dem() -> np.ndarray:
    cols = [c for c, _ in _CHAIN_PROFILE]
    elev = [e for _, e in _CHAIN_PROFILE]
    profile = np.interp(np.arange(_CHAIN_W, dtype=float), cols, elev)
    return np.tile(profile, (_CHAIN_H, 1))


def _chain_transform():
    return from_bounds(*_CHAIN_BOUNDS, width=_CHAIN_W, height=_CHAIN_H)


def _chain_seeds() -> list[BasinSeed]:
    west, _south, _east, north = _CHAIN_BOUNDS
    return [
        BasinSeed(
            name=name,
            lon=west + (_CHAIN_SEED_COL[name] + 0.5) * _CHAIN_CELL_DEG,
            lat=north - (_CHAIN_SEED_ROW + 0.5) * _CHAIN_CELL_DEG,
            level_m=_CHAIN_LEVELS[name],
        )
        for name in ("caspian", "pontic", "mediterranean")
    ]


def _chain_expect(value_m: float, lon: float) -> SillExpectation:
    """A chain-world crest, stated from the profile that defines it."""
    return SillExpectation(
        value_m=value_m,
        frame=_SYNTH_FRAME,
        source="tests/test_viz.py::_CHAIN_PROFILE — a control point of the profile",
        measured_by="closed form: np.interp over the profile's own control points",
        lon=lon,
        lat=_CHAIN_BOUNDS[3] - (_CHAIN_SEED_ROW + 0.5) * _CHAIN_CELL_DEG,
    )


#: The chain world's three pairs.  Two crests are control points of the profile;
#: the far pair has no common border at all (the pontic basin lies bodily between
#: them), so it is DECLARED unmeasured rather than given a number.
def _chain_checks() -> dict:
    lat_col = lambda c: _CHAIN_BOUNDS[0] + (c + 0.5) * _CHAIN_CELL_DEG  # noqa: E731
    return {
        ("caspian", "pontic"): _chain_expect(
            _CHAIN_CREST_CASPIAN_PONTIC_M, lat_col(60)
        ),
        ("mediterranean", "pontic"): _chain_expect(
            _CHAIN_CREST_PONTIC_MED_M, lat_col(120)
        ),
        ("caspian", "mediterranean"): UnmeasuredSill(
            "the pontic basin lies bodily between them, so these two territories "
            "share no border and there is no saddle to measure at any resolution"
        ),
    }


class TestSpillIsADonorLevelFloodNotAPairwiseMax:
    """Connectivity must never flood an intermediate basin to the donor's level."""

    def test_a_high_donor_does_not_connect_through_an_intermediate_basin(self):
        """The whole three-basin hero frame in one assertion.

        caspian spills into pontic across the +20 m crest, and STOPS there: the
        water it delivers joins the pontic water body at the pontic's own -30 m
        surface, which is 35 m below the +5 m crest to the mediterranean.  Under
        the pairwise-max rule this frame reported TWO spilling pairs and
        ``corridor='ramp'`` raised rather than render.
        """
        field = assign_basins(
            _chain_dem(), _chain_transform(), _chain_seeds(),
            sill_expectations=_chain_checks(), dem_frame=_SYNTH_FRAME,
            sill_tolerance_m=0.5,
        )
        assert field.spilling is True
        assert field.connected_groups == (("caspian", "pontic"), ("mediterranean",))
        assert field.corridor_mode == "ramp"

    def test_the_verdict_does_not_depend_on_the_order_the_basins_are_listed(self):
        seeds = _chain_seeds()
        dem, xform = _chain_dem(), _chain_transform()
        checks = _chain_checks()
        reference = assign_basins(
            dem, xform, seeds, sill_expectations=checks,
            dem_frame=_SYNTH_FRAME, sill_tolerance_m=0.5,
        )
        for order in itertools.permutations(range(3)):
            field = assign_basins(
                dem, xform, [seeds[i] for i in order], sill_expectations=checks,
                dem_frame=_SYNTH_FRAME, sill_tolerance_m=0.5,
            )
            assert field.spilling == reference.spilling
            assert field.connected_groups == reference.connected_groups
            assert field.sill_m.keys() == reference.sill_m.keys()
            for pair, expected in reference.sill_m.items():
                # NaN == NaN is False, and "this saddle was never reached" is a
                # verdict that has to survive a re-ordering just like a number.
                assert field.sill_m[pair] == pytest.approx(expected, nan_ok=True)

    def test_the_uninvolved_basin_keeps_its_own_water_surface(self):
        """A basin that is neither donor nor receiver must not be repainted by the
        corridor ramp.  The mediterranean stands at -45 m; the ramp interpolates
        between +35 m and -30 m and would put it 15 m too high.
        """
        field = assign_basins(
            _chain_dem(), _chain_transform(), _chain_seeds(),
            sill_expectations=_chain_checks(), dem_frame=_SYNTH_FRAME,
            sill_tolerance_m=0.5,
        )
        med = field.basin_id == (field.names.index("mediterranean") + 1)
        assert med.any()
        assert np.allclose(field.water_level_m[med], _CHAIN_LEVELS["mediterranean"])
        assert not field.corridor_mask[med].any()

    def test_the_two_crests_are_measured_and_the_far_pair_never_touches(self):
        field = assign_basins(
            _chain_dem(), _chain_transform(), _chain_seeds(),
            sill_expectations=_chain_checks(), dem_frame=_SYNTH_FRAME,
            sill_tolerance_m=0.5,
        )
        assert field.sill_m[("caspian", "pontic")] == pytest.approx(
            _CHAIN_CREST_CASPIAN_PONTIC_M, abs=0.5
        )
        assert field.sill_m[("mediterranean", "pontic")] == pytest.approx(
            _CHAIN_CREST_PONTIC_MED_M, abs=0.5
        )
        # The pontic basin lies bodily between them, so these two territories have
        # no common border and there is no saddle to report.
        assert np.isnan(field.sill_m[("caspian", "mediterranean")])

    def test_a_pontic_high_enough_to_clear_its_own_crest_does_connect_onward(self):
        """The rule must not be "never transitive" -- it must be "each hop on its
        own level".  Raise the pontic above the +5 m crest and the three become one
        connected group through TWO genuine corridors (which ramp mode then refuses,
        by design, rather than guessing how the discharge divides).
        """
        seeds = [
            BasinSeed(s.name, s.lon, s.lat, 10.0 if s.name == "pontic" else s.level_m)
            for s in _chain_seeds()
        ]
        field = assign_basins(
            _chain_dem(), _chain_transform(), seeds, corridor="step",
            sill_expectations=_chain_checks(), dem_frame=_SYNTH_FRAME,
            sill_tolerance_m=0.5,
        )
        assert field.connected_groups == (("caspian", "mediterranean", "pontic"),)


# ---------------------------------------------------------------------------
# The seam blend must not move a sill
# ---------------------------------------------------------------------------
#
# ``build_epoch_mosaic`` used to Gaussian-blur a strip of +/-3 sigma around EVERY
# integer degree line inside the mosaic, whether or not a tile boundary was there.
# On the single-file inputs both bakes actually pass, that is pure interior
# smoothing, and measured on the real working raster
# (``gebco_2024_khvalynian_pontic_...cog.tif``, 2026-08-01) it moved the Manych
# point sill at 44.119 E / 45.773 N by
#
#     max_px=2000 (the Pontic bake's own setting) : +24.00 m -> +40.76 m
#     max_px=4000                                 : +25.00 m -> +30.57 m
#
# and changed 41-75 % of all mosaic cells by more than a centimetre.  A +16.76 m
# rise turns a Caspian spilling over a +23.00 m sill at a +35 m highstand into an
# endorheic lake -- silently, and in the direction that deletes the subject of the
# animation.

_MANYCH_LON, _MANYCH_LAT = 44.119, 45.773
_MANYCH_SILL_M = 23.00
_MANYCH_BOUNDS = (43.0, 45.0, 45.5, 46.5)      # west, south, east, north
_MANYCH_W, _MANYCH_H = 160, 96                 # -> 0.015625 deg/px, both axes
_MANYCH_ROW, _MANYCH_COL = 46, 71              # the cell holding the sill point
_MANYCH_SHOULDER_M = 80.0


def _manych_dem() -> np.ndarray:
    """A narrow E-W channel with an analytic crest at the real Manych point.

    The channel is three cells tall -- the Manych spillway really is narrower than
    a 0.015625 deg working pixel is comfortable with -- and its floor is a tent of
    slope 6 m/px peaking at exactly ``_MANYCH_SILL_M`` in cell
    ``(_MANYCH_ROW, _MANYCH_COL)``, which is the cell containing 44.119 E /
    45.773 N.  Nothing here is measured off a picture.
    """
    dem = np.full((_MANYCH_H, _MANYCH_W), _MANYCH_SHOULDER_M, dtype="float32")
    cols = np.arange(_MANYCH_W)
    floor = np.maximum(_MANYCH_SILL_M - 6.0 * np.abs(cols - _MANYCH_COL), -60.0)
    dem[_MANYCH_ROW - 1 : _MANYCH_ROW + 2, :] = floor.astype("float32")
    return dem


class TestSeamBlendDoesNotMoveTheManychCrest:
    @pytest.fixture()
    def manych_tile(self, tmp_path: Path) -> Path:
        p = tmp_path / "manych_fusion.tif"
        _write_geotiff(p, _manych_dem(), bounds=_MANYCH_BOUNDS)
        return p

    def test_the_sill_point_really_is_the_crest_of_the_raw_raster(self):
        """Guard the fixture itself: the number the mosaic is pinned against has
        to be a property of the raster, not of the test's arithmetic."""
        dem = _manych_dem()
        assert dem[_MANYCH_ROW, _MANYCH_COL] == pytest.approx(_MANYCH_SILL_M)
        channel = dem[_MANYCH_ROW - 1 : _MANYCH_ROW + 2, :]
        assert channel.max() == pytest.approx(_MANYCH_SILL_M)
        # The sill point really is where the fixture says it is.
        xform = from_bounds(*_MANYCH_BOUNDS, width=_MANYCH_W, height=_MANYCH_H)
        col_f, row_f = ~xform * (_MANYCH_LON, _MANYCH_LAT)
        assert (int(row_f), int(col_f)) == (_MANYCH_ROW, _MANYCH_COL)

    def test_a_single_file_mosaic_reproduces_the_manych_crest_exactly(
        self, manych_tile: Path
    ):
        """A single file has no seams, so the mosaic must BE the raster."""
        cfg = RenderConfig(max_px=_MANYCH_W)          # native resolution, no resample
        dem, _ice, _xf, _bnds, _crs, _res = build_epoch_mosaic(
            {"m": str(manych_tile)}, cfg
        )
        assert dem.shape == (_MANYCH_H, _MANYCH_W)
        assert dem[_MANYCH_ROW, _MANYCH_COL] == pytest.approx(_MANYCH_SILL_M, abs=1e-4)
        np.testing.assert_allclose(dem, _manych_dem(), rtol=0, atol=1e-4)

    def test_the_legacy_degree_grid_blend_is_what_moved_the_crest(
        self, manych_tile: Path
    ):
        """The defect, pinned as a measurement rather than described in prose.

        43-45 E and 45-46 N all cross this window, the sill point sits 7.6 px from
        the 44 E line and 14.5 px from the 46 N line, and sigma=5 px reaches 15 px
        -- so the old rule blurred the shoulders straight into the spillway.
        """
        cfg = RenderConfig(max_px=_MANYCH_W, legacy_degree_grid_blend=True)
        with pytest.warns(SeamBlendShiftWarning):
            dem, *_ = build_epoch_mosaic({"m": str(manych_tile)}, cfg)
        risen = float(dem[_MANYCH_ROW, _MANYCH_COL])
        # Far more than the 2 m the project's own SillResolutionError tolerates.
        assert risen > _MANYCH_SILL_M + 5.0

    def test_a_zero_seam_sigma_no_longer_crashes(self, manych_tile: Path):

_SEAM_COL = 110                                # -> 44.71875 E, 39 px east of the sill
_SEAM_LON = _MANYCH_BOUNDS[0] + _SEAM_COL * 0.015625
_SEAM_STEP_M = 12.0                            # the tile disagreement a blend hides


def _manych_two_tile_raw() -> np.ndarray:
    """The Manych window as two tiles that disagree by ``_SEAM_STEP_M`` at the seam.

    A seam blend exists to hide a step like this.  It is put 39 px east of the
    sill point so that "blend the seam" and "leave the sill alone" are separable
    claims rather than the same claim.
    """
    raw = _manych_dem().copy()
    raw[:, _SEAM_COL:] += np.float32(_SEAM_STEP_M)
    return raw


class TestSeamBlendIsSeamLocalAndAudited:
    @pytest.fixture()
    def abutting_tiles(self, tmp_path: Path) -> list[Path]:
        """Two tiles sharing the 44.71875 E edge; the Manych channel crosses both.

        The shared edge is deliberately NOT a whole degree: a seam is where the
        tiles are.  The old rule could not see one that failed to land on a whole
        degree any more than it could avoid one that was not there at all.
        """
        raw = _manych_two_tile_raw()
        a = tmp_path / "left.tif"
        b = tmp_path / "right.tif"
        _write_geotiff(a, raw[:, :_SEAM_COL], bounds=(43.0, 45.0, _SEAM_LON, 46.5))
        _write_geotiff(b, raw[:, _SEAM_COL:], bounds=(_SEAM_LON, 45.0, 45.5, 46.5))
        return [a, b]

    def test_the_seam_is_found_from_the_tiles_not_from_the_degree_grid(
        self, abutting_tiles: list[Path]
    ):
        from rasterio.coords import BoundingBox

        tiles = [BoundingBox(43.0, 45.0, _SEAM_LON, 46.5),
                 BoundingBox(_SEAM_LON, 45.0, 45.5, 46.5)]
        lons, lats = interior_tile_seams(
            tiles, BoundingBox(43.0, 45.0, 45.5, 46.5), 0.015625
        )
        assert lons == [pytest.approx(_SEAM_LON)]
        assert lats == []          # both tiles span the full latitude range
        # A single tile has no interior edge at all, whatever its extent.
        assert interior_tile_seams(
            [BoundingBox(43.0, 45.0, 45.5, 46.5)],
            BoundingBox(43.0, 45.0, 45.5, 46.5),
            0.015625,
        ) == ([], [])

    def test_a_genuine_seam_is_still_blended_but_only_near_the_seam(
        self, abutting_tiles: list[Path]
    ):
        cfg = RenderConfig(max_px=_MANYCH_W)
        with pytest.warns(SeamBlendShiftWarning):
            dem, *_ = build_epoch_mosaic({str(p): str(p) for p in abutting_tiles}, cfg)
        raw = _manych_two_tile_raw()
        hw = int(np.ceil(cfg.seam_sigma_px * 3))
        # Blended where the tiles meet ...
        assert not np.allclose(
            dem[:, _SEAM_COL - 2 : _SEAM_COL + 2],
            raw[:, _SEAM_COL - 2 : _SEAM_COL + 2],
        )
        # ... and untouched everywhere else, the Manych crest included.
        np.testing.assert_allclose(
            dem[:, : _SEAM_COL - hw], raw[:, : _SEAM_COL - hw], rtol=0, atol=1e-4
        )
        np.testing.assert_allclose(
            dem[:, _SEAM_COL + hw + 1 :],
            raw[:, _SEAM_COL + hw + 1 :],
            rtol=0,
            atol=1e-4,
        )
        assert dem[_MANYCH_ROW, _MANYCH_COL] == pytest.approx(_MANYCH_SILL_M, abs=1e-4)

    def test_the_shift_a_blend_causes_is_reported_with_a_number_and_a_place(
        self, abutting_tiles: list[Path]
    ):
        cfg = RenderConfig(max_px=_MANYCH_W)

            build_epoch_mosaic({str(p): str(p) for p in abutting_tiles}, cfg)
        message = str(record[0].message)
        assert " m at " in message and " E / " in message and " N" in message
        assert "seam_blend_max_shift_m" in message

    def test_a_blend_that_stays_inside_its_bound_says_nothing(self, tmp_path: Path):

        b = tmp_path / "flat_right.tif"

        _write_geotiff(b, flat[:, _SEAM_COL:], bounds=(_SEAM_LON, 45.0, 45.5, 46.5))

# ===========================================================================
# T1 -- the sill guard is MANDATORY, and what it accepts is PROVENANCED
# ===========================================================================
#
# Until 2026-08-03 ``assign_basins`` looped over ``(sill_expectations or {})``, so
# the default ``None`` validated nothing at any number of seeds: the guard existed,
# was documented, was tested -- and never ran unless a caller opted in.  Neither
# bake opted in.  The tests below pin the refusals that replace it, and they are
# written against the SYNTHETIC worlds so they run in the ordinary suite; the
# production numbers they protect are pinned by the dataset tests further down.


class TestTheSillGuardIsMandatory:
    def test_multi_seed_without_expectation_raises(self):
        """The headline defect: two basins, no expectation, and a shipped frame."""
        dem = _basin_dem()
        with pytest.raises(MissingSillExpectationError) as exc:
            assign_basins(dem, _basin_transform(), _seeds(-40.0, -60.0))
        message = str(exc.value)
        assert "east<->west" in message
        # It has to name what the caller must supply, not merely complain.
        assert "SillExpectation" in message and "UnmeasuredSill" in message

    def test_a_single_basin_frame_needs_no_expectation(self):
        """One basin has no pair, so there is nothing to check and nothing to say.

        The guard must not become a tax on the frames it does not protect.
        """
        lon, lat = _rc_to_lonlat(_A_RC)
        field = assign_basins(
            _one_bowl_dem(), _basin_transform(), [BasinSeed("west", lon, lat, -40.0)]
        )
        assert field.sill_m == {}
        assert field.spilling is False

    def test_a_single_basin_frame_still_validates_what_it_IS_given(self):
        """A6: the structural checks used to sit inside ``if pair_keys:``.

        With one seed there are no pairs, so the whole block was skipped and every
        supplied expectation — a bare float, a key naming basins the frame does not
        have, a frame-mismatched one — was swallowed without a word.  Only the
        COMPARISON needs a pair.
        """
        lon, lat = _rc_to_lonlat(_A_RC)
        one = [BasinSeed("west", lon, lat, -40.0)]
        dem, xform = _one_bowl_dem(), _basin_transform()

        # A bare float is a bare float at any number of basins.
        with pytest.raises(SillExpectationTypeError):
            assign_basins(
                dem, xform, one, dem_frame=_SYNTH_FRAME,
                sill_expectations={("east", "west"): 27.701},
            )
        # ... and so is a key that is not a pair of this call.  The message says
        # WHY there is no pair, rather than leaving the caller to infer it.
        with pytest.raises(ValueError, match="ONE basin"):
            assign_basins(
                dem, xform, one, dem_frame=_SYNTH_FRAME,
                sill_expectations={("east", "west"): _expect(_SADDLE_M)},
            )
        # A string key would be split into characters by ``tuple(sorted(...))``.
        with pytest.raises(ValueError, match="not a pair of basin names"):
            assign_basins(
                dem, xform, one, dem_frame=_SYNTH_FRAME,
                sill_expectations={"west": _expect(_SADDLE_M)},
            )

    def test_the_same_pair_under_both_orderings_is_refused(self):
        """A5: ``checks[tuple(sorted(raw_pair))] = given`` overwrote in silence.

        Two entries are two supplied MEASUREMENTS.  Keeping whichever dict
        iteration order reached last discards one of them and never mentions it —
        a value decided by iteration order passing as a measurement, which is this
        guard's own stated defect class.  Reproduced below with a -999 m and a
        +5 m claim about the same saddle: before the fix the call returned
        normally, having compared exactly one of the two.
        """
        dem = gaussian_filter(_basin_dem(), sigma=1.5)
        with pytest.raises(ValueError) as exc:
            assign_basins(
                dem, _basin_transform(), _seeds(-40.0, -60.0),
                dem_frame=_SYNTH_FRAME,
                sill_expectations={
                    ("east", "west"): _expect(-999.0),
                    ("west", "east"): _expect(5.0),
                },
            )
        message = str(exc.value)
        assert "twice, under both orderings" in message
        assert "east<->west" in message
        # Both discarded claims are named, so neither can be lost silently.
        assert "-999" in message and "5.0" in message
        # It is not a SillResolutionError: the point is that the comparison must
        # not happen at all until the caller says which claim is theirs.
        assert not isinstance(exc.value, SillResolutionError)

    def test_bare_float_expectation_is_refused(self):
        """A number with no frame, no source and no place is indistinguishable
        downstream from a measured one -- which is the whole defect class."""
        dem = _basin_dem()
        with pytest.raises(SillExpectationTypeError) as exc:
            assign_basins(
                dem,
                _basin_transform(),
                _seeds(-40.0, -60.0),
                dem_frame=_SYNTH_FRAME,
                sill_expectations={("east", "west"): 27.701},
            )
        assert isinstance(exc.value, TypeError)
        assert "float" in str(exc.value)

    def test_partial_expectation_coverage_is_refused(self):
        """Naming one pair of three leaves the other two as unchecked as None did."""
        dem = _chain_dem()
        checks = _chain_checks()
        del checks[("mediterranean", "pontic")]
        with pytest.raises(MissingSillExpectationError, match="mediterranean<->pontic"):
            assign_basins(
                dem, _chain_transform(), _chain_seeds(),
                sill_expectations=checks, dem_frame=_SYNTH_FRAME,
            )

    def test_an_unmeasured_sill_needs_a_written_reason(self):
        """Abstaining is allowed; abstaining silently is the thing being removed."""
        with pytest.raises(ValueError, match="written reason"):
            UnmeasuredSill("   ")

    def test_an_expectation_needs_a_frame_a_source_and_a_method(self):
        for field_name in ("frame", "source", "measured_by"):
            kwargs = {
                "value_m": 27.701, "frame": "F0", "source": "s",
                "measured_by": "m", "lon": 44.1, "lat": 45.7,
            }
            kwargs[field_name] = ""
            with pytest.raises(ValueError, match=field_name):
                SillExpectation(**kwargs)

    def test_a_non_finite_expectation_cannot_certify_a_frame_by_accident(self):
        with pytest.raises(ValueError, match="value_m"):
            SillExpectation(
                value_m=float("nan"), frame="F0", source="s",
                measured_by="m", lon=44.1, lat=45.7,
            )

    def test_an_unmeasured_pair_is_recorded_not_hidden(self):
        """The refusal has to be visible afterwards, or it is just silence again."""
        from paleoeurope.viz.paleocostline_render import _basin_title_line

        dem = _basin_dem()
        field = _assign2(
            dem, _seeds(-40.0, -60.0),
            expect_m=None,
            unmeasured_because="no independent survey of this synthetic crest",
        )
        status = field.sill_expectation_status[("east", "west")]
        assert status.startswith("UNMEASURED: ")
        assert "no independent survey" in status
        assert field.sill_expected_m[("east", "west")] is None
        # ... and it reaches the caption, not only the sidecar.
        assert "UNVERIFIED sill: east~west" in _basin_title_line(field)


class TestTheExpectationFrameMustMatchTheRaster:
    """Comparing an F0 constant against an F1 saddle is the error A5 made.

    It was wrong by exactly ``dz_sill_m`` = +2.701 m at 16.5 ka, and it produced
    the (refuted) claim that the guard could not be satisfied at any resolution.

    def test_expectation_frame_must_match_raster(self):
        dem = gaussian_filter(_basin_dem(), sigma=1.5)
        wrong_frame = SillExpectation(
            value_m=_SADDLE_M - 4.79,
            frame="F0 present-day datum (H_modern + dz_solid)",
            source="a frame this fixture is not in",
            measured_by="irrelevant -- the frame check runs before the comparison",
            lon=_SADDLE_LON,
            lat=_SADDLE_LAT,
        )
        with pytest.raises(SillFrameMismatchError) as exc:
            assign_basins(

                dem_frame=_SYNTH_FRAME,
                sill_expectations={("east", "west"): wrong_frame},
            )
        assert "F0 present-day datum" in str(exc.value)
        assert _SYNTH_FRAME in str(exc.value)

    def test_a_value_that_would_otherwise_pass_still_fails_on_the_frame(self):
        """The frame check is not a formality reachable only when the number is
        already wrong: this expectation agrees to 0.00 m and is still refused."""
        dem = gaussian_filter(_basin_dem(), sigma=1.5)
        agreeing = _expect(_SADDLE_M - 4.79)
        # Same number, correctly framed, is accepted:
        _assign2(dem, _seeds(-40.0, -60.0), expect_m=_SADDLE_M - 4.79)
        mislabelled = SillExpectation(
            value_m=agreeing.value_m, frame="F1", source=agreeing.source,
            measured_by=agreeing.measured_by, lon=agreeing.lon, lat=agreeing.lat,
        )
        with pytest.raises(SillFrameMismatchError):
            assign_basins(
                dem, _basin_transform(), _seeds(-40.0, -60.0),
                dem_frame=_SYNTH_FRAME,
                sill_expectations={("east", "west"): mislabelled},
            )

    def test_an_undeclared_dem_frame_is_a_refusal_not_a_pass(self):
        """A check that cannot run must not look like a check that passed."""
        dem = gaussian_filter(_basin_dem(), sigma=1.5)
        with pytest.raises(SillFrameMismatchError, match="no dem_frame"):
            assign_basins(
                dem, _basin_transform(), _seeds(-40.0, -60.0),
                sill_expectations={("east", "west"): _expect(_SADDLE_M - 4.79)},
            )

    def test_the_frame_is_read_from_the_raster_not_from_the_caller(
        self, tmp_path: Path
    ):
        tagged = tmp_path / "tagged.tif"
        _write_geotiff(tagged, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS)
        untagged = tmp_path / "untagged.tif"
        _write_geotiff(
            untagged, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS,
            vertical_frame=None,
        )
        assert declared_vertical_frame({"a": str(tagged)}) == _SYNTH_FRAME
        assert declared_vertical_frame({"a": str(untagged)}) is None
        # Two tiles that disagree cannot be merged into one frame.
        other = tmp_path / "other.tif"
        _write_geotiff(
            other, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS,
            vertical_frame="F1 palaeo water surface",
        )
        with pytest.raises(ValueError, match="different vertical_frame"):
            declared_vertical_frame({"a": str(tagged), "b": str(other)})

    def test_silence_from_some_tiles_is_not_agreement(self, tmp_path: Path):
        """A7: the docstring promised it, the code did the opposite.

        The loop skipped an untagged tile and returned the one tag it found, so a
        mosaic in which ONE tile of two declares a frame reported that frame for the
        WHOLE merged surface.  That is the same defect as returning None for a
        tagged raster, with the sign reversed — and it is worse, because the value
        returned is then compared against a SillExpectation and passes.
        """
        tagged = tmp_path / "tagged.tif"
        _write_geotiff(tagged, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS)
        untagged = tmp_path / "untagged.tif"
        _write_geotiff(
            untagged, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS,
            vertical_frame=None,
        )
        # Each on its own is unambiguous ...
        assert declared_vertical_frame({"a": str(tagged)}) == _SYNTH_FRAME
        assert declared_vertical_frame({"a": str(untagged)}) is None
        # ... and NOTHING declaring a frame is still None: there is no claim to
        # contradict, and the caller is refused downstream instead.
        assert declared_vertical_frame(
            {"a": str(untagged), "b": str(untagged)}
        ) is None
        # ... but a mixed mosaic has no stated frame at all.
        with pytest.raises(ValueError) as exc:
            declared_vertical_frame({"a": str(tagged), "b": str(untagged)})
        assert "SILENCE AS AGREEMENT" in str(exc.value)
        assert "declares nothing" in str(exc.value)
        assert "untagged.tif" in str(exc.value)

    def test_a_path_that_does_not_exist_is_not_agreement_either(
        self, tmp_path: Path
    ):
        """``build_epoch_mosaic`` silently DROPS a non-existent input, so the
        mosaic that gets built is not the one the caller described.  Certifying a
        frame for it would certify something nobody asked for."""
        tagged = tmp_path / "tagged.tif"
        _write_geotiff(tagged, _basin_dem().astype("float32"), bounds=_BASIN_BOUNDS)
        with pytest.raises(ValueError, match="does not exist"):
            declared_vertical_frame(
                {"a": str(tagged), "b": str(tmp_path / "never_baked.tif")}
            )
        # With nothing tagged there is still nothing to contradict.
        assert declared_vertical_frame({"b": str(tmp_path / "never_baked.tif")}) is None


class TestTheToleranceIsDerivedNotBorrowed:
    """The 2 m was justified in a circle: RenderConfig.seam_blend_max_shift_m cited
    assign_basins' sill_tolerance_m, whose docstring cited it back, and
    build_epoch_mosaic repeated the loop a third time.  Both now default to one
    named constant whose comment derives it from the spill window."""

    def test_both_defaults_come_from_the_one_derived_constant(self):
        import inspect

        from paleoeurope.viz import paleocostline_render as mod

        assert SILL_TOLERANCE_M == 2.0
        assert RenderConfig().seam_blend_max_shift_m == SILL_TOLERANCE_M
        assert (
            inspect.signature(assign_basins).parameters["sill_tolerance_m"].default
            == SILL_TOLERANCE_M
        )
        source = inspect.getsource(mod)
        head = source[: source.index("class RenderConfig")]
        # The derivation is a duration converted to metres, and it says so.
        assert "0.3 ka" in head and "0.25 ka" in head
        assert "MANYCH_SILL_PATH_M" in head

    def test_the_derivation_does_not_contradict_its_own_arithmetic(self):
        """A11.  The comment converted 2.00 m to ~0.3 ka of spill window and then
        called it "the largest error that cannot change how many frames a viewer
        counts as spilling".  The grid is 0.25 ka and 0.3 > 0.25, so 2 m CAN move
        the count by one frame: the claim was the negation of its own arithmetic.

        The brief (§4, "Tolerance") says keep 2.0 m and re-derive it in writing, so
        what is pinned here is that the writing now states the consequence instead
        of denying it.
        """
        import inspect

        from paleoeurope.viz import paleocostline_render as mod

        source = inspect.getsource(mod)
        head = source[: source.index("class RenderConfig")]
        assert SILL_TOLERANCE_M == 2.0
        # The retracted claim survives EXACTLY ONCE, and only as a quotation
        # inside its own retraction -- the project's convention is that the wrong
        # sentence and its correction stay side by side rather than one of them
        # vanishing from the record.  What must not exist is an unqualified
        # restatement of it.
        assert head.count("cannot change how many frames") == 1
        after = head[head.index("cannot change how many frames"):]
        assert "which is what this comment used to claim" in after[:400]
        assert "arithmetic contradicts" in after[:400]
        # What replaced it: one frame, not zero, and why not two.
        assert "one frame" in head
        assert "0.5 ka" in head
        # And the alternative (re-deriving down to 1.67 m) is recorded as
        # considered and declined, not left as an unexplained gap.
        assert "1.67" in head

        )

        # The renderer must read the config field, not the sill tolerance.
        renderer = inspect.getsource(mod.render_paleocostline_epoch)
        assert "smoothing_shift_max_m=cfg.smoothing_shift_disclose_m" in renderer
        assert "smoothing_shift_max_m=basin_sill_tolerance_m" not in renderer

        assert "MANYCH_SILL_M" in doc and "lower bound" in doc


# ===========================================================================
# T4 -- what is drawn, what is refused, and how the difference is disclosed
# ===========================================================================
#
# A channel with ONE controlling crest, a stretch of bed that stands above the
# ramped water surface, an isolated pocket below it, and a second dry stretch
# before the receiving basin.  Every elevation is a control point of the profile.
#
# This is the production failure reduced to its bones: on khv_16.5ka.tif at
# max_px=7800 the same geometry leaves the drawn Caspian and the drawn Pontic as
# two separate bodies while the terrain verdict is "spilling", and hands the
# stranded pocket -- there, the Aegean -- a surface between the two basin levels
# that no basin in the frame carries.

_POCKET_H, _POCKET_W = 30, 200
_POCKET_BOUNDS = (0.0, 0.0, 20.0, 3.0)
_POCKET_CELL_DEG = 0.1
_POCKET_ROW = 15
_POCKET_HALF = 4
_POCKET_CREST_M = 55.0
_POCKET_PROFILE = (
    (0, 300.0), (4, -80.0), (40, -80.0),              # donor bowl
    (58, _POCKET_CREST_M), (62, _POCKET_CREST_M),     # THE crest -- the sill
    (68, 52.0), (80, 52.0),                           # bed above the ramp -> dry
    (90, 10.0), (105, 10.0),                          # the stranded pocket
    (115, 5.0), (125, 5.0),                           # a second dry stretch
    (140, -60.0), (195, -60.0), (199, 300.0),         # receiver bowl
)
_POCKET_LEVELS = (60.0, -20.0)                        # donor, receiver


def _pocket_dem() -> np.ndarray:
    cols = [c for c, _ in _POCKET_PROFILE]
    elev = [e for _, e in _POCKET_PROFILE]
    profile = np.interp(np.arange(_POCKET_W, dtype=float), cols, elev)
    dem = np.full((_POCKET_H, _POCKET_W), 300.0)
    # The channel is 9 cells tall, not 3: the renderer smooths with sigma=1.5 px
    # before it assigns basins, and a 3-cell channel between +300 m walls has its
    # floor pulled up ~70 m by that Gaussian -- which would make this fixture a
    # test of the smoothing rather than of the corridor.
    dem[_POCKET_ROW - _POCKET_HALF : _POCKET_ROW + _POCKET_HALF + 1, :] = profile
    return dem


def _pocket_transform():
    return from_bounds(*_POCKET_BOUNDS, width=_POCKET_W, height=_POCKET_H)


def _pocket_seeds() -> list[BasinSeed]:
    west, _s, _e, north = _POCKET_BOUNDS
    lat = north - (_POCKET_ROW + 0.5) * _POCKET_CELL_DEG
    return [
        BasinSeed("west", west + 20.5 * _POCKET_CELL_DEG, lat, _POCKET_LEVELS[0]),
        BasinSeed("east", west + 175.5 * _POCKET_CELL_DEG, lat, _POCKET_LEVELS[1]),
    ]


def _pocket_checks(
    value_m: float | None = _POCKET_CREST_M, *, reason: str = ""
) -> dict:
    key = ("east", "west")
    if value_m is None:
        return {key: UnmeasuredSill(reason)}
    return {
        key: SillExpectation(
            value_m=value_m,
            frame=_SYNTH_FRAME,
            source=(
                "tests/test_viz.py::_POCKET_PROFILE — a control point, not a reading"
            ),
            measured_by="closed form: the single highest control point on the route",
            lon=_POCKET_BOUNDS[0] + 60.5 * _POCKET_CELL_DEG,
            lat=_POCKET_BOUNDS[3] - (_POCKET_ROW + 0.5) * _POCKET_CELL_DEG,
        )
    }


def _pit_dem() -> np.ndarray:
    """The pocket world with a closed pit inside the RECEIVER's territory.

    Below the receiver's own level and cut off from its water by the +10 / +5 m
    stretches on either side, so it is painted at the receiver's declared level
    while being a water body of its own that nothing in the frame measured.
    """
    dem = _pocket_dem().copy()
    dem[_POCKET_ROW - _POCKET_HALF : _POCKET_ROW + _POCKET_HALF + 1, 108:112] = -30.0
    return dem


class TestWaterNoBasinAccountsFor:
    def test_unseeded_water_body_inside_extent_raises(self):
        """A body on neither end of the corridor, painted anyway."""
        with pytest.raises(UnseededWaterBodyError) as exc:
            assign_basins(
                _pocket_dem(), _pocket_transform(), _pocket_seeds(),
                sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
            )
        message = str(exc.value)
        # It lists the components and their sizes rather than just objecting.
        assert "cells at" in message and "water bodies totalling" in message
        # And it names the seeded levels, so the reader can see which surfaces the
        # frame does state and that this body is not one of them.
        assert "west +60.00 m" in message and "east -20.00 m" in message
        # The way out is a provenanced seed, NOT a change of corridor mode: the
        # message used to offer corridor='step', which is where the guard could not
        # fire at all.
        assert "seed each body with a provenanced level" in message
        assert "do not switch corridor mode" in message

    def test_the_refusal_fires_in_step_mode_too(self):
        """A2, the structural hole.

        In ``corridor="step"`` the surface array is ``level_lookup[basin_id]``, so
        every painted cell holds a declared level BY CONSTRUCTION: the old test —
        "is this body's surface one of the declared levels" — had exactly one
        possible answer and the guard built on it could not fire.  Step is the mode
        the acceptance artifact is rendered in, so on the shipped native frame the
        Aegean+Marmara and the Levantine were still painted, wearing the Pontic
        lake's surface.
        """
        # First: prove the premise on this fixture.  Every painted surface IS a
        # declared level in step mode, so a float comparison would find nothing.
        with pytest.raises(UnseededWaterBodyError) as exc:
            assign_basins(
                _pit_dem(), _pocket_transform(), _pocket_seeds(), corridor="step",
                sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
            )
        message = str(exc.value)
        # The body wears the receiver's own declared level -20.00 m and is refused
        # anyway, because the level a frame declares is not a water body it
        # declares.
        assert "-20.00…-20.00 m" in message
        assert "component MEMBERSHIP" in message
        assert "corridor='ramp' and corridor='step' alike" in message

    def test_the_same_pit_is_refused_in_ramp_mode(self):
        """Both modes, one rule — the finding's "it must fire in both"."""
        with pytest.raises(UnseededWaterBodyError):
            assign_basins(
                _pit_dem(), _pocket_transform(), _pocket_seeds(), corridor="ramp",
                sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
            )

    def test_a_float_comparison_could_not_have_caught_it(self):
        """Pins the premise rather than asserting it in prose.

        Painting the pit at the receiver's level is invisible to any test of the
        SURFACE values: in step mode the set of painted surfaces is a subset of the
        declared levels, exactly. The defect is which BODY wears one.
        """
        # Seeding the pit is the way out the refusal names, and it also gives us a
        # field to inspect: with every body seeded, the frame renders.
        west, _s, _e, north = _POCKET_BOUNDS
        lat = north - (_POCKET_ROW + 0.5) * _POCKET_CELL_DEG
        seeds = _pocket_seeds() + [
            BasinSeed("pit", west + 109.5 * _POCKET_CELL_DEG, lat, -25.0)
        ]
        # With the pit seeded it sits BETWEEN the other two in a 1-D channel, so
        # east and west no longer share a border at all and no pair here has a
        # closed-form crest this fixture is about.
        checks = {
            pair: UnmeasuredSill(
                "this fixture exercises the unseeded-body guard, not this saddle"
            )
            for pair in (("east", "west"), ("east", "pit"), ("pit", "west"))
        }
        field = assign_basins(
            _pit_dem(), _pocket_transform(), seeds, corridor="step",
            sill_expectations=checks, dem_frame=_SYNTH_FRAME,
        )
        painted = set(
            np.unique(field.water_level_m[np.isfinite(field.water_level_m)]).tolist()
        )
        declared = set(field.levels_m)
        assert painted <= declared        # <- the old guard's whole test, satisfied
        assert field.water_mask.any()

    def test_step_mode_invents_nothing(self):
        field = assign_basins(
            _pocket_dem(), _pocket_transform(), _pocket_seeds(), corridor="step",
            sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
        )
        surfaces = set(
            np.unique(field.water_level_m[np.isfinite(field.water_level_m)]).tolist()
        )
        assert surfaces <= set(_POCKET_LEVELS)
        assert field.spilling is True


class TestTheMeasuredRouteIsKeptAndDrawn:
    """``_priority_flood_ownership`` measured a minimax route and threw the
    predecessors away, so the module could state a sill it could not show."""

    def test_the_route_is_a_real_path_whose_highest_cell_is_the_sill(self):
        field = assign_basins(
            _pocket_dem(), _pocket_transform(), _pocket_seeds(), corridor="step",
            sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
        )
        route = field.sill_route_rc[("east", "west")]
        assert len(route) > 100
        # 4-connected, no jumps.
        steps = {
            (abs(a[0] - b[0]), abs(a[1] - b[1])) for a, b in zip(route, route[1:])
        }
        assert steps <= {(0, 1), (1, 0)}
        dem = _pocket_dem()
        along = np.array([dem[rc] for rc in route])
        assert along.max() == pytest.approx(field.sill_m[("east", "west")], abs=1e-9)
        # It runs from one seed's cell to the other's, not from the crest outwards.
        ends = {route[0], route[-1]}
        seed_cells = set()
        for seed in _pocket_seeds():
            col_f, row_f = ~_pocket_transform() * (seed.lon, seed.lat)
            seed_cells.add((int(np.floor(row_f)), int(np.floor(col_f))))
        assert ends == seed_cells

    def test_a_saddle_above_the_ceiling_yields_no_route_rather_than_a_short_one(self):
        field = _assign2(
            _basin_dem(), _seeds(-40.0, -60.0),
            expect_m=None,
            unmeasured_because="the fixture caps the flood below the saddle",
            ceiling_margin_m=10.0,
        )
        assert field.sill_route_rc[("east", "west")] == ()


class TestTheRouteIsActuallyDrawn:
    """The polyline is the disclosure; a sidecar field nobody plots is not one."""

    def _field(self):
        return assign_basins(
            _pocket_dem(), _pocket_transform(), _pocket_seeds(), corridor="step",
            sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
        )

    def test_the_route_reaches_the_figure_with_a_legend_entry(self):
        import matplotlib.pyplot as plt

        from paleoeurope.viz.paleocostline_render import _draw_sill_routes

        field = self._field()
        fig, ax = plt.subplots()
        try:
            handle = _draw_sill_routes(ax, field, _pocket_transform(), RenderConfig())
            assert handle is not None
            assert "NOT water" in handle.get_label()
            assert handle.get_label() == SILL_ROUTE_LEGEND_LABEL
            # One polyline plus one crest marker.
            assert len(ax.lines) == 2
            drawn_lon, drawn_lat = ax.lines[0].get_data()
            assert len(drawn_lon) == len(field.sill_route_rc[("east", "west")])
            # It is drawn in map coordinates, inside the raster's own extent.
            west, south, east, north = _POCKET_BOUNDS
            assert west <= min(drawn_lon) and max(drawn_lon) <= east
            assert south <= min(drawn_lat) and max(drawn_lat) <= north
            # The crest marker sits at the measured sill position.
            assert ax.lines[1].get_data()[0][0] == pytest.approx(
                field.sill_lonlat[("east", "west")][0]
            )
            # The annotation states the elevation the verdict rests on.
            texts = [t.get_text() for t in ax.texts]
            assert any("sill +" in t for t in texts)
        finally:
            plt.close(fig)

    def test_it_can_be_switched_off_without_switching_off_the_measurement(self):
        import matplotlib.pyplot as plt

        from paleoeurope.viz.paleocostline_render import _draw_sill_routes

        field = self._field()
        fig, ax = plt.subplots()
        try:
            assert _draw_sill_routes(
                ax, field, _pocket_transform(), RenderConfig(draw_sill_route=False)
            ) is None
            assert len(ax.lines) == 0
        finally:
            plt.close(fig)
        # The route is still measured and still in the field.
        assert len(field.sill_route_rc[("east", "west")]) > 100


class TestTheDrawnRouteDoesNotClaimADeterminateSpillway:
    """A4.  The polyline IS a valid minimax path — 4-connected, endpoints at the
    two seeds, ``max(dem)`` along it equal to the reported sill to the last
    float64 bit (pinned by ``TestTheMeasuredRouteIsKeptAndDrawn``).  What it is
    NOT is *the* spillway: a minimax path is an equivalence class and the member
    that gets drawn is whatever the priority flood's flat-index tie-break
    produced.  On the real raster that member runs hundreds of km north of the
    Manych-Don-Azov line and dives through the abyssal Black Sea.

    Legending an arbitrary representative "Measured spill route (minimax path
    through the sill)" is the picture asserting a route the data does not
    determine.  These tests fail if the claim comes back.
    """

    #: Phrases that assert a determinate spillway.  ``the`` + a route noun is the
    #: whole defect: the definite article is what turns one member of a class into
    #: the answer.
    DETERMINACY_CLAIMS = (
        "measured spill route",
        "the spill route",
        "the spillway",
        "the minimax path",
        "the measured route",
        "the route through",
    )

    def test_the_legend_does_not_claim_determinacy(self):
        label = SILL_ROUTE_LEGEND_LABEL.lower()
        for claim in self.DETERMINACY_CLAIMS:
            assert claim not in label, claim
        # And it says the true thing out loud, in the legend itself, not only in a
        # sidecar nobody opens.
        assert "one minimax path" in label
        assert "not unique" in label
        assert "not a mapped spillway" in label

    def test_the_sidecar_note_says_which_part_the_data_determines(self):
        note = SILL_ROUTE_NOTE
        for claim in self.DETERMINACY_CLAIMS:
            assert claim not in note.lower(), claim
        assert "ONE minimax path" in note
        assert "not a determinate spillway" in note
        assert "equivalence class" in note
        # It names what IS determinate, so "not unique" cannot be read as "not
        # measured": the crest elevation and its cell are the measurement.
        assert "elevation_m" in note and "crest" in note
        # ... and it states the specific way the drawn line misleads on the real
        # raster, with MEASURED figures rather than only conceding non-uniqueness in
        # the abstract.  Each of them is re-derived by
        # TestTheProductionFrameOnRealTerrain::
        # test_the_drawn_route_is_a_valid_minimax_path_and_not_a_spillway.
        assert "abyssal Black Sea" in note
        for token in ("10,095-cell", "47.4854 N", "1.750 deg", "195 km",
                      "45.7354 N", "1,193", "11.8 %", "-2,509.77 m"):
            assert token in note, token

    def test_the_crest_annotation_says_the_path_to_it_is_not(self):
        import matplotlib.pyplot as plt

        from paleoeurope.viz.paleocostline_render import _draw_sill_routes

        field = assign_basins(
            _pocket_dem(), _pocket_transform(), _pocket_seeds(), corridor="step",
            sill_expectations=_pocket_checks(), dem_frame=_SYNTH_FRAME,
        )
        fig, ax = plt.subplots()
        try:
            _draw_sill_routes(ax, field, _pocket_transform(), RenderConfig())
            texts = [t.get_text() for t in ax.texts]
            assert any("crest determinate" in t for t in texts)
            assert any("the path drawn to it is not" in t for t in texts)
        finally:
            plt.close(fig)

    def test_the_note_reaches_every_sill_entry_in_the_sidecar(self, tmp_path: Path):
        import json

        csv = tmp_path / "spratt.csv"
        _write_spratt_csv(
            csv, [(ka * 1000, float(-ka * 5), 2.0) for ka in range(0, 27)]
        )
        tile = tmp_path / "pocket_fusion.tif"
        _write_geotiff(tile, _pocket_dem().astype("float32"), bounds=_POCKET_BOUNDS)
        png = render_paleocostline_epoch(
            epoch_ka=16.5,
            tile_paths={str(tile): str(tile)},
            sea_level_csv=csv,
            paleocoastlines_shp=tmp_path / "nonexistent.shp",
            output_dir=tmp_path / "out",
            config=RenderConfig(
                max_px=_POCKET_W, dpi=40, fig_size_in=4.0, seam_sigma_px=0.1
            ),
            basin_seeds=_pocket_seeds(),
            basin_corridor="step",
            basin_sill_expectations=_pocket_checks(None, reason="not the subject"),
        )
        meta = json.loads(

        )
        assert meta["sills"]
        for sill in meta["sills"]:

# ---------------------------------------------------------------------------
# A1 -- the one sentence the whole task-4 deliverable rests on
# ---------------------------------------------------------------------------
#
# It is written verbatim into every .basins.json and pointed at from the figure
# title.  The version it replaces said the channel "runs below the interpolated
# water surface for part of its length, and those cells are left dry rather than
# painted".  Both halves were wrong:
#
#   * SIGN.  A cell below the water surface is exactly the cell that gets PAINTED
#     -- ``dem_sm < water_level`` is the entire test.  The cells are dry because
#     the BED IS ABOVE the surface.  The stated cause was the negation of the real
#     one.
#   * ALGORITHM.  "the interpolated water surface" describes something
#     corridor="step" does not do, and step is the mode the shipped artifact is
#     rendered in.
#
# A sentence shipped in a sidecar is data.  This class pins it.

_PINNED_STEP_DISCLOSURE = (
    "The spill verdict is measured on the TERRAIN, not read off the blue in this "
    "image: it is the connected-path minimax between the basin seeds — the lowest "
    "crest a continuous channel can be held under — compared against the donor "
    "basin's own declared surface. The blue answers a different question, cell by "
    "cell: a cell is painted only where the bed lies BELOW the water surface that "
    "cell carries. This frame is corridor='step': every cell is drawn at its OWN "
    "basin's flat declared level and nothing is interpolated anywhere. Along the "
    "spill route the bed stands AT OR ABOVE the level of the basin that owns it "
    "for part of its length, so those cells are left dry — they are dry BECAUSE "
    "the ground there is higher than the water surface, which is the opposite of "
    "what an earlier version of this sentence said. Painting the corridor anyway "
    "was costed and refused: on the native 16.5 ka frame it drew 1,099,766 cells "
    "/ 167,305 km² over bed as high as +203.71 m, and it made the two basins look "
    "joined. So two separate blue shapes under the word 'spilling' are not a "
    "contradiction and not a near-miss: the crest is passable at the donor's "
    "level, and the channel floor beyond it still stands above the level the "
    "receiving side is drawn at. The verdict is about a crest; the paint is about "
    "every cell. Neither one is a correction of the other, and the basins did NOT "
    "visibly merge here — if they had, painted_water.basins_joined would say "
    "true. The orange polyline is ONE minimax path through that crest (it is not "
    "the only one — see sills[].route_note); how many of its cells are drawn dry "
    "is painted_water.gap_cells, out of painted_water.route_cells_by_pair."
)

class TestTheSmoothingNoteIsPerPairNotPerFrame:
    """A12.  ``sill_shift`` starts as ``dict.fromkeys(sill_m, None)`` and is filled
    only where the RAW flood also reached that saddle, but the note beside it was
    chosen from one frame-wide ``smoothing_measured`` flag -- so a pair the
    unsmoothed flood never reached shipped ``null`` under a sentence reading
    "measured, NEVER subtracted".
    """

    def test_the_four_reasons_are_told_apart(self):
        from paleoeurope.viz.paleocostline_render import _smoothing_shift_note

        measured = _smoothing_shift_note(
            shift=0.25, measured_here=True, smoothing_measured=True
        )
        assert "measured on THIS pair" in measured

        off = _smoothing_shift_note(
            shift=None, measured_here=True, smoothing_measured=False
        )
        assert off.startswith("NOT MEASURED")
        assert "measure_smoothing_sill_shift is False" in off

        unreachable = _smoothing_shift_note(
            shift=None, measured_here=False, smoothing_measured=True
        )
        assert "never reached the pair's saddle" in unreachable

        raw_missed = _smoothing_shift_note(
            shift=None, measured_here=True, smoothing_measured=True
        )
        assert "these two territories never met" in raw_missed
        # None of the three refusals may read as a measurement, and none may read
        # as a zero.
        for note in (off, unreachable, raw_missed):
            assert "measured on THIS pair" not in note
            assert "Not 0.0" in note or "not 0.0" in note or "reported as 0.0" in note

    def test_a_pair_the_raw_flood_never_reached_is_not_called_measured(self):
        """Measured on the fixture rather than asserted about the helper.

        The two-bowl saddle is +40.00 m raw and ~+35.5 m after sigma=1.5.  A flood
        ceiling of +38 m therefore reaches it on the SMOOTHED array and not on the
        raw one, so ``sill_m`` is finite while ``sill_smoothing_shift_m`` is None --
        exactly the state that used to be annotated "measured".
        """
        from paleoeurope.viz.paleocostline_render import _smoothing_shift_note

        raw = _basin_dem()
        smoothed = gaussian_filter(raw, sigma=1.5)
        pair = ("east", "west")
        field = assign_basins(
            smoothed, _basin_transform(), _seeds(-40.0, -60.0),
            sill_expectations=_two_bowl_checks(_SADDLE_M - 4.79),
            dem_frame=_SYNTH_FRAME,
            unsmoothed_dem=raw,
            sill_tolerance_m=2.0,
            ceiling_margin_m=78.0,       # ceiling = -40 + 78 = +38 m
        )
        assert field.ceiling_m == pytest.approx(38.0)
        assert np.isfinite(field.sill_m[pair])          # the smoothed grid sees it
        assert field.sill_smoothing_shift_m[pair] is None   # the raw one does not
        note = _smoothing_shift_note(
            shift=field.sill_smoothing_shift_m[pair],
            measured_here=bool(np.isfinite(field.sill_m[pair])),
            smoothing_measured=True,
        )
        assert note.startswith("NOT MEASURED for this pair")
        assert "these two territories never met" in note

#
# They need Y:\paleo and D:\pasha\RESEARCH\renders on disk, ~7.7 GB of committed

# ---------------------------------------------------------------------------
# The seeds these tests use, and what is and is not claimed for them
# ---------------------------------------------------------------------------
#
# NOT a provenanced registry.  RAISE-not-compute item 6 (a BasinSeed must come
# from a registry recording box + rule + raster hash, in the DATA) is NOT closed
# by this file and must not be read as closed: these two coordinates are the
# argmin of a box chosen by an agent, and a lon/lat without the grid it was
# derived on can only be trusted, not re-derived.
#
# They are usable HERE because the quantity these tests assert -- the
# connected-path saddle -- was measured to be seed-independent on this raster:
# 1953 seed pairs (31 Caspian x 63 Pontic, every tile-argmin below -200 m) all
# return the same float64 +28.000944137573242 and the same crest cell, at all
# three resolutions (B1-MEASURE, 2026-08-03).  What is NOT seed-independent is the
# corridor and the painted water -- moving the Pontic seed from the Black Sea to
# Marmara changes the corridor by 2,383,005 cells -- so no test below asserts a
# corridor extent.
_KHV_CASPIAN_SEED = (50.74375, 38.95208)
_KHV_PONTIC_SEED = (35.28542, 42.20208)
#: From the raster's own ``threshold_defensible`` tag: T = level(+27.00 m F2)
#: + dz(+2.944 m over the named locus) = +29.944 m.  Read, not re-derived.
_KHV_CASPIAN_LEVEL_M = 29.944
#: Contract 5a-bis's per-row T for 16.5 ka -- the one Pontic row whose marker
#: (the Bosphorus sill) is real, bar +/-1.2 m.  No other epoch has one.
_KHV_PONTIC_LEVEL_M = -18.020


def _khv_seeds() -> list[BasinSeed]:
    return [
        BasinSeed("caspian", *_KHV_CASPIAN_SEED, level_m=_KHV_CASPIAN_LEVEL_M),
        BasinSeed("pontic", *_KHV_PONTIC_SEED, level_m=_KHV_PONTIC_LEVEL_M),
    ]


def _khv_expectation() -> SillExpectation:
    """The Manych expectation, built FROM THE DATA rather than from a constant.

    Quantity: ``sill_path_f1_m`` = ``MANYCH_SILL_PATH_M`` (+25.00 m, the
    continuous-path crest of the modern GEBCO raster) + ``dz_sill_m`` (+2.701 m,
    the epoch's ICE-6G displacement at the sill) = **+27.701 m**.

    NOT ``MANYCH_SILL_M`` (+23.00 m): ``bake_khvalynian`` documents that constant
    as the max of per-column minima, i.e. a LOWER BOUND on the path crest.  Armed
    as the expectation it false-fails by +2.30 m on the native grid, where the
    frame and the terrain in fact agree to +0.30 m.
    """
    frame = declared_vertical_frame({"khv": str(_KHV_TIF)})
    df = pd.read_csv(_KHV_SILL_CSV, comment="#")
    row = df[(df["epoch_ka"] == 16.5) & (df["scenario"] == "defensible")].iloc[0]
    return SillExpectation(
        value_m=float(row["sill_path_f1_m"]),
        frame=frame,
        source=(
            f"{_KHV_SILL_CSV.name} row epoch_ka=16.5 scenario=defensible, column "
            f"sill_path_f1_m = MANYCH_SILL_PATH_M +25.00 (F0, modern GEBCO) + "
            f"dz_sill_m {float(row['dz_sill_m']):+.6f} (ICE-6G_C at 16.5 ka)"
        ),
        measured_by=(
            "connected-path minimax (priority flood) over the modern GEBCO "
            "15-arcsec COG at native resolution, then the epoch's dz added at the "
            "sill — NOT a single-cell reading of the sill point, which is a "
            "different quantity by up to 8 m on this raster"
        ),
        # The crest cell that flood returns; re-derived by
        # test_manych_path_constant_matches_a_native_flood.
        lon=44.10208333333333,
        lat=45.70625,
    )


@pytest.fixture(scope="module")
def khv_native():
    """The native Khvalynian mosaic and the array the mask is built from.

    Module-scoped because building it is 3 s and holding it is ~400 MB, while the
    priority flood over it is 70-80 s and every test below needs the same grid.
    """
    from paleoeurope.viz.paleocostline_render import build_epoch_mosaic

    dem, _ice, xform, _bounds, _crs, _res = build_epoch_mosaic(
        {"khv": str(_KHV_TIF)}, RenderConfig(max_px=7800)
    )
    nodata = np.isnan(dem)
    dem_sm = gaussian_filter(np.where(nodata, 0.0, dem), sigma=1.5)
    dem_sm = np.where(nodata, np.nan, dem_sm)
    yield {
        "dem": dem,
        "dem_sm": dem_sm,
        "xform": xform,
        "nodata": nodata,
        "frame": declared_vertical_frame({"khv": str(_KHV_TIF)}),
    }


def _native_saddle(
    dem: np.ndarray, xform, seeds: list[BasinSeed] | None = None
) -> dict:
    """The caspian<->pontic saddle AND its drawn route, from ONE priority flood.

    Called instead of ``assign_basins`` wherever the test's subject is the NUMBER
    rather than the guard.  ``assign_basins`` now refuses this seed pair on the
    real raster — the Aegean+Marmara and the Levantine are painted water bodies
    holding no seed (see ``UnseededWaterBodyError``) — so a test that only wants
    the crest must measure it, not render it.  Same flood, same tie-break, same
    float: these are the two functions ``assign_basins`` itself calls for it.
    """
    from paleoeurope.viz.paleocostline_render import (
        _measure_sills,
        _minimax_route,
        _priority_flood_ownership,
    )

    dem = np.asarray(dem, dtype=np.float64)
    seeds = _khv_seeds() if seeds is None else seeds
    seed_rc = []
    for seed in seeds:
        col_f, row_f = ~xform * (seed.lon, seed.lat)
        seed_rc.append((int(np.floor(row_f)), int(np.floor(col_f))))
    ceiling = max(s.level_m for s in seeds) + 100.0
    basin_id, bottleneck, parent = _priority_flood_ownership(dem, seed_rc, ceiling)
    value, (row, col), partner = _measure_sills(basin_id, bottleneck, 2)[(1, 2)]
    route = _minimax_route(parent, (row, col), partner)
    rows = np.array([rc[0] for rc in route])
    cols = np.array([rc[1] for rc in route])
    route_lon, route_lat = xform * (cols + 0.5, rows + 0.5)
    lon, lat = xform * (col + 0.5, row + 0.5)
    return {
        "saddle": float(value),
        "crest": (float(lon), float(lat)),
        "route": route,
        "route_lon": np.asarray(route_lon),
        "route_lat": np.asarray(route_lat),
        "route_elev": dem[rows, cols],
        "seed_rc": seed_rc,
    }


@pytest.fixture(scope="module")
def khv_native_saddle(khv_native):
    """The dem_sm saddle, one flood, shared by every test that needs the number."""
    return _native_saddle(khv_native["dem_sm"], khv_native["xform"])


@pytest.fixture(scope="module")
def khv_raw_saddle(khv_native):
    """The same measurement on the UNSMOOTHED mosaic, one flood."""
    return _native_saddle(
        np.where(khv_native["nodata"], np.nan, khv_native["dem"]),
        khv_native["xform"],
    )


@_dataset
@pytest.mark.integration
@pytest.mark.slow
class TestTheTwoBasinFrameIsRenderableOnlyAtNativeResolution:
    """C2, as a refusal rather than a knob.

    The connected-path saddle on ``dem_sm``, against the bake's own +27.701 m:

        max_px=2000  -> +36.005, gap +8.30   REFUSED
        max_px=4000  -> +32.942, gap +5.24   REFUSED
        max_px=7800  -> +28.001, gap +0.30   accepted

    Nothing in the module raises ``max_px`` on the caller's behalf, and no
    minimum-``max_px`` precondition is enforced: resolution is not monotone in
    sill fidelity (2048 px resolves the POINT cell 1.00 m worse than 2000 px), so
    such a rule would be a heuristic with a measured blind spot.
    """

    @pytest.mark.parametrize(
        "max_px, measured, gap",
        [(2000, "+36.01", "+8.30"), (4000, "+32.94", "+5.24")],
    )
    def test_two_basin_frame_refuses_below_native_resolution(
        self, tmp_path: Path, max_px: int, measured: str, gap: str
    ):
        # The eustatic curve is read for the figure LABEL on every path, including
        # this one, so it has to exist even though no basin level comes from it.
        curve = tmp_path / "spratt.csv"
        _write_spratt_csv(
            curve, [(ka * 1000, float(-ka * 5), 2.0) for ka in range(0, 27)]
        )
        with pytest.raises(SillResolutionError) as exc:
            render_paleocostline_epoch(
                epoch_ka=16.5,
                tile_paths={"khv": str(_KHV_TIF)},
                sea_level_csv=curve,
                paleocoastlines_shp=tmp_path / "nonexistent.shp",
                output_dir=tmp_path / f"px{max_px}",
                config=RenderConfig(max_px=max_px, dpi=40, fig_size_in=4.0),
                basin_seeds=_khv_seeds(),
                basin_corridor="step",
                basin_sill_expectations={
                    ("caspian", "pontic"): _khv_expectation()
                },
            )
        message = str(exc.value)
        assert measured in message
        assert "+27.70" in message
        assert gap in message
        # The refusal has to say why raising max_px is not offered as a fix.
        assert "REFUSAL, NOT A SETTING" in message
        assert "not monotone" in message.lower()
        assert not list((tmp_path / f"px{max_px}").glob("*.png"))

    def test_at_native_resolution_the_same_expectation_is_accepted(
        self, khv_native_saddle
    ):
        """The number, measured off the same flood ``assign_basins`` runs.

        It is measured directly rather than through ``assign_basins`` because that
        call now REFUSES this seed pair on this raster — not on the sill, which
        agrees to +0.30 m, but on two painted seas that hold no seed.  The two
        facts are separated so neither can hide the other; the refusal itself is
        pinned by
        ``test_the_native_two_seed_frame_passes_the_sill_guard_and_is_refused_on_the_water``.
        """
        measured = khv_native_saddle["saddle"]
        expected = _khv_expectation().value_m
        assert measured == pytest.approx(28.000944137573242, abs=1e-9)
        assert measured - expected == pytest.approx(0.2999, abs=0.001)
        assert abs(measured - expected) < SILL_TOLERANCE_M

    def test_the_native_two_seed_frame_passes_the_sill_guard_and_is_refused_on_the_water(
        self, khv_native
    ):
        """The guard order, pinned: sill first, water second.

        ``UnseededWaterBodyError`` — not ``SillResolutionError`` — proves the
        expectation was accepted, because the sill comparison runs before the water
        surface is built and would have raised first had it disagreed.
        """
        with pytest.raises(UnseededWaterBodyError) as exc:
            assign_basins(
                khv_native["dem_sm"],
                khv_native["xform"],
                _khv_seeds(),
                corridor="step",
                sill_expectations={("caspian", "pontic"): _khv_expectation()},
                dem_frame=khv_native["frame"],
            )
        assert not isinstance(exc.value, SillResolutionError)
        assert "179 water bodies totalling 717,981 cells" in str(exc.value)

    def test_the_guarded_array_is_the_one_the_mask_is_built_from(
        self, khv_native_saddle, khv_raw_saddle
    ):
        """C2's other half: guarding ``dem`` while painting ``dem_sm`` certifies a
        surface nobody draws.  The two differ by +0.25 m at native and +3.30 m at
        max_px=2000 -- and at 2000 px that difference is more than the tolerance."""
        smoothed = khv_native_saddle["saddle"]
        raw = khv_raw_saddle["saddle"]
        assert raw == pytest.approx(27.751062393188477, abs=1e-9)
        assert smoothed == pytest.approx(28.000944137573242, abs=1e-9)
        # dem_sm is +0.250 m higher, i.e. the smoothing moves the crest UP -- the
        # direction that turns a spilling basin endorheic.  This is the number the
        # sidecar used to carry as sill[].smoothing_shift_m on the native frame.
        assert smoothed - raw == pytest.approx(0.249882, abs=1e-5)

    def test_the_lower_bound_constant_would_false_fail_here(self, khv_native):
        """C1, pinned: ``MANYCH_SILL_M`` = +23.00 m is a LOWER BOUND on the path
        crest, and arming it as the expectation refuses a frame that is right."""
        lower_bound = SillExpectation(
            value_m=23.00 + 2.701019287109375,
            frame=khv_native["frame"],
            source="MANYCH_SILL_M +23.00 (max of per-column minima) + dz_sill_m",
            measured_by="a LOWER BOUND on the path crest, not the path crest",
            lon=44.119,
            lat=45.773,
        )
        with pytest.raises(SillResolutionError, match=r"\+2\.30"):
            assign_basins(
                khv_native["dem_sm"], khv_native["xform"], _khv_seeds(),
                corridor="step",
                sill_expectations={("caspian", "pontic"): lower_bound},
                dem_frame=khv_native["frame"],
            )

    def test_the_native_two_seed_frame_is_refused_rather_than_painting_two_unseeded_seas(
        self, tmp_path: Path
    ):
        """RAISE-not-compute item 5, on the frame the artifact is rendered from.

        This test used to assert that the native two-seed frame RENDERS, and it
        did: ``corridor="step"`` writes ``level_lookup[basin_id]``, so the old
        "is the surface a declared level" test could not fire, and the frame
        shipped with the Aegean+Marmara (535,508 cells, ~92,400 km2) and the
        Levantine (77,232 cells) painted at the Pontic lake's -18.02 m — a surface
        neither of them has, ~2.4x the 38,040 km2 of spurious drowning that
        contract 5a exists to remove.

        With the guard on the quantity that matters, the frame is REFUSED.  That is
        the intended outcome and it is also a live consequence: **the two-seed
        native artifact cannot be produced as specified.**  What would produce it is
        named in the refusal and is not inventable here — a provenanced level for
        the Aegean+Marmara and one for the Levantine, seeded like any other basin.
        Substituting a plausible number for either is the defect class this whole
        phase exists to remove.

        ``measure_smoothing_sill_shift=False`` only to avoid a second 75 s flood

                output_dir=outdir,
                config=RenderConfig(
                    max_px=7800, dpi=100, ice_model_label="ICE-6G_C",
                    measure_smoothing_sill_shift=False,
                ),
                data_source_label="khvalynian-defensible",
                basin_seeds=_khv_seeds(),
                basin_corridor="step",
                basin_sill_expectations={("caspian", "pontic"): _khv_expectation()},

        message = str(exc.value)
        # In step mode every one of these wears a DECLARED level, which is exactly
        # why the old surface-comparison could not see them.
        assert "179 water bodies totalling 717,981 cells" in message
        assert "535,508 cells at -18.02…-18.02 m" in message   # Aegean + Marmara
        assert "77,232 cells at -18.02…-18.02 m" in message    # Levantine
        assert "component MEMBERSHIP" in message
        assert "corridor='ramp' and corridor='step' alike" in message
        # A refused frame writes nothing: no half-PNG, no sidecar claiming a state.
        assert not list(outdir.glob("*.png"))
        assert not list(outdir.glob("*.basins.json"))

    def test_the_drawn_route_is_a_valid_minimax_path_and_not_a_spillway(
        self, khv_native_saddle
    ):
        """A4, both halves, measured on the production raster.

        Every figure in ``SILL_ROUTE_NOTE`` is re-derived here, because that
        sentence ships inside every ``.basins.json`` and a sentence in a sidecar is
        data.  It has to be as checkable as the floats beside it.
        """
        got = khv_native_saddle
        route, lat, elev = got["route"], got["route_lat"], got["route_elev"]

        # HALF ONE -- the line is a genuine minimax path, so "not unique" must not
        # be read as "not measured".
        steps = {
            (abs(a[0] - b[0]), abs(a[1] - b[1])) for a, b in zip(route, route[1:])
        }
        assert steps <= {(0, 1), (1, 0)}                       # 4-connected
        assert {route[0], route[-1]} == set(got["seed_rc"])    # seed to seed
        assert float(np.nanmax(elev)) == got["saddle"]         # bit-exact crest
        assert got["saddle"] == pytest.approx(28.000944137573242, abs=1e-9)
        assert len(route) == 10095

        # HALF TWO -- and it is nowhere near a mapped spillway, which is why the
        # legend may not call it "the" spill route.
        assert got["crest"] == pytest.approx((44.1104, 45.7354), abs=1e-3)
        assert float(lat.max()) == pytest.approx(47.4854, abs=1e-3)
        north_deg = float(lat.max()) - got["crest"][1]
        assert north_deg == pytest.approx(1.750, abs=1e-3)
        assert north_deg * 111.32 == pytest.approx(194.8, abs=0.5)
        deep = elev < -1000.0
        assert int(deep.sum()) == 1193
        assert 100.0 * float(deep.mean()) == pytest.approx(11.8, abs=0.1)
        assert float(np.nanmin(elev)) == pytest.approx(-2509.77, abs=0.01)

        # The sentence the sidecar ships carries exactly these figures.
        for token in ("10,095-cell", "47.4854 N", "1.750 deg", "195 km",
                      "45.7354 N", "1,193", "11.8 %", "-2,509.77 m"):
            assert token in SILL_ROUTE_NOTE, token

    def test_manych_path_constant_matches_a_native_flood(self):
        """Pins ``MANYCH_SILL_PATH_M`` = +25.00 m as a re-derivable measurement.

        A1 measured it once on the modern GEBCO COG; this re-runs the flood, so
        the constant the expectation is built from is not taken on trust.

        Measured through ``_priority_flood_ownership`` / ``_measure_sills`` rather
        than ``assign_basins``: this test IS the measurement of a crest, and the
        modern COG at a 0 m Pontic level carries closed depressions holding no seed,
        which ``assign_basins`` now (correctly) refuses to paint.  The flood, the
        tie-break and the float are the same either way — ``assign_basins`` calls
        these two functions for exactly this number.
        """
        from paleoeurope.viz.paleocostline_render import build_epoch_mosaic

        dem, _ice, xform, _b, _c, _r = build_epoch_mosaic(
            {"cog": str(_MODERN_COG)}, RenderConfig(max_px=7800)
        )
        got = _native_saddle(
            np.asarray(dem, dtype=np.float64),
            xform,
            seeds=[
                BasinSeed("caspian", *_KHV_CASPIAN_SEED, level_m=-28.0),
                BasinSeed("pontic", *_KHV_PONTIC_SEED, level_m=0.0),
            ],
        )
        assert got["saddle"] == pytest.approx(25.00, abs=0.05)
        assert got["crest"] == pytest.approx((44.1021, 45.7062), abs=0.001)
        # ... and it is the coordinate the production expectation carries.
        expectation = _khv_expectation()
        assert (expectation.lon, expectation.lat) == pytest.approx(
            got["crest"], abs=1e-9
        )
        # +25.00 (F0) + dz_sill_m +2.701 = the expectation's own value.
        df = pd.read_csv(_KHV_SILL_CSV, comment="#")
        row = df[(df["epoch_ka"] == 16.5) & (df["scenario"] == "defensible")].iloc[0]
        assert got["saddle"] + float(row["dz_sill_m"]) == pytest.approx(
            expectation.value_m, abs=1e-9
        )


@_dataset
@pytest.mark.integration
@pytest.mark.slow
def test_pontic_single_basin_path_is_byte_stable(tmp_path: Path):
    """The 39-frame invariant: nothing A changed may move the single-basin path.

    Re-renders 16.5 ka gradual with W0's own recipe and compares sha256 against the
    file W0 left on disk.  A stored hash would only prove this file has not been
    edited; re-rendering proves the CODE still produces it.
    """
    import hashlib
    import importlib.util

    spec = importlib.util.spec_from_file_location(
        "_bbs_for_bytestability", _REPO_ROOT / "scripts" / "bake_black_sea.py"
    )
    bbs = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = bbs
    spec.loader.exec_module(bbs)

    paleo_tif = _REPO_ROOT / "outputs" / "_bs_paleo" / "bs_16.5ka.tif"
    if not (paleo_tif.exists() and _W0_PONTIC_FRAME.exists()):
        pytest.skip("the W0 Pontic tree / cached paleo-DEM is not on this machine")

    curve = pd.read_csv(bbs.PONTIC_CSV, comment="#")
    threshold = bbs.pontic_threshold(curve, 16.5, "gradual")
    out = render_paleocostline_epoch(
        16.5,
        {"bs": str(paleo_tif)},
        sea_level_csv=bbs.SPRATT_CSV,
        paleocoastlines_shp=tmp_path / "nonexistent.shp",
        output_dir=tmp_path / "bytestable",
        config=RenderConfig(max_px=2000, ice_model_label="ICE-6G_C"),
        data_source_label=f"Pontic-gradual-{threshold.rule}",
        sea_level_override_m=threshold.threshold_m,
        force_dem_mask=True,
    )
    assert hashlib.sha256(out.read_bytes()).hexdigest() == hashlib.sha256(
        _W0_PONTIC_FRAME.read_bytes()
    ).hexdigest()
    # The single-basin path writes no basin sidecar and never enters the guard.
    assert not (out.parent / (out.stem + ".basins.json")).exists()
